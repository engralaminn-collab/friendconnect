   154→function formatBytes(bytes: number | null | undefined): string {
   155→  if (bytes === null || bytes === undefined) return ''
   156→  if (bytes === 0) return '0 B'
   157→  const units = ['B', 'KB', 'MB', 'GB']
   158→  let n = bytes
   159→  let i = 0
   160→  while (n >= 1024 && i < units.length - 1) {
   161→    n /= 1024
   162→    i++
   163→  }
   164→  return `${n.toFixed(n < 10 && i > 0 ? 1 : 0)} ${units[i]}`
   165→}

   167→function fileTypeFromMime(mime: string | null | undefined): MessageType {
   168→  if (!mime) return 'document'
   169→  if (mime.startsWith('image/')) return 'image'
   170→  if (mime.startsWith('video/')) return 'video'
   171→  if (mime.startsWith('audio/')) return 'audio'
   172→  return 'document'
   173→}

   175→function linkify(text: string): React.ReactNode[] {
   176→  const parts = text.split(/(https?:\/\/[^\s]+)/g)
   177→  return parts.map((part, i) => {
   178→    if (/^https?:\/\//.test(part)) {
   179→      return (
   180→        <a key={i} href={part} target="_blank" rel="noreferrer" className="underline break-all">
   181→          {part}
   182→        </a>
   183→      )
   184→    }
   185→    return <span key={i}>{part}</span>
   186→  })
   187→}

   189→function toISO(d: any): string {
   190→  if (!d) return new Date().toISOString()
   191→  if (typeof d === 'string') return d
   192→  if (d instanceof Date) return d.toISOString()
   193→  try {
   194→    return new Date(d).toISOString()
   195→  } catch {
   196→    return new Date().toISOString()
   197→  }
   198→}

   204→function normalizeMessage(raw: any): Message {
   205→  return {
   206→    id: raw.id,
   207→    senderId: raw.senderId,
   208→    senderName: raw.senderName ?? raw.sender?.name ?? null,
   209→    senderPhoto: raw.senderPhoto ?? raw.sender?.profile?.photoUrl ?? null,
   210→    receiverId: raw.receiverId ?? null,
   211→    groupId: raw.groupId ?? null,
   212→    content: raw.content ?? '',
   213→    type: (raw.type ?? 'text') as MessageType,
   214→    fileUrl: raw.fileUrl ?? null,
   215→    fileName: raw.fileName ?? null,
   216→    fileSize: raw.fileSize ?? null,
   217→    mimeType: raw.mimeType ?? null,
   218→    status: (raw.status ?? 'SENT') as MessageStatus,
   219→    createdAt: toISO(raw.createdAt),
   220→    reactions: Array.isArray(raw.reactions) ? raw.reactions.map((r: any) => ({ emoji: r.emoji, userId: r.userId, userName: r.userName })) : [],
   221→    deletedFor: raw.deletedFor ?? '',
   222→    forwardedFrom: raw.forwardedFrom ?? null,
   223→    starred: !!raw.starred,
   224→    expiresAt: raw.expiresAt ?? null,
   225→  }
   226→}

   228→// A message is rendered as the "🚫 This message was deleted" placeholder when
   229→// it was deleted for everyone OR explicitly deleted-for-me (deletedFor is a
   230→// comma-separated list of user ids, or the literal 'everyone').
   231→function isDeletedForMe(msg: Message, meId?: string): boolean {
   232→  if (!msg.deletedFor) return false
   233→  if (msg.deletedFor === 'everyone') return true
   234→  if (!meId) return false
   235→  return msg.deletedFor
   236→    .split(',')
   237→    .map((s) => s.trim())
   238→    .filter(Boolean)
   239→    .includes(meId)
   240→}

   242→function lastMessagePreview(c: Conversation, meId: string | undefined): string {
   243→  if (!c.lastMessage) return 'Tap to start chatting'
   244→  const lm = c.lastMessage
   245→  const prefix = meId && lm.senderId === meId ? 'You: ' : ''
   246→  if (lm.type === 'image') return `${prefix}📷 Photo`
   247→  if (lm.type === 'video') return `${prefix}🎥 Video`
   248→  if (lm.type === 'audio') return `${prefix}🎤 Audio`
   249→  if (lm.type === 'document') return `${prefix}📄 Document`
   250→  return `${prefix}${lm.content}`
   251→}

   270→// ---------------- conversation list ----------------
   271→function ConversationList() {
   272→  const [tab, setTab] = useState<'chats' | 'groups'>('chats')
   273→  const [query, setQuery] = useState('')
   274→  const [conversations, setConversations] = useState<Conversation[]>([])
   275→  const [groups, setGroups] = useState<GroupLite[]>([])
   276→  const [loading, setLoading] = useState(true)
   277→  const { openChat, openGroupChat } = useApp()
   278→  const { socket, isOnline } = useSocket()
   279→  const { user: me } = useAuth()
   280→  const meId = me?.id
   281→
   282→  const load = useCallback(async () => {
   283→    setLoading(true)
   284→    try {
   285→      const [c, g] = await Promise.all([
   286→        api<{ conversations: Conversation[] }>('/api/messages/conversations').catch(() => ({ conversations: [] as Conversation[] })),
   287→        api<{ groups: GroupLite[] }>('/api/groups').catch(() => ({ groups: [] as GroupLite[] })),
   288→      ])
   289→      setConversations(c.conversations)
   290→      setGroups(g.groups)
   291→    } catch (e: any) {
   292→      toast.error(e?.message || 'Failed to load')
   293→    } finally {
   294→      setLoading(false)
   295→    }
   296→  }, [])
   297→
   298→  useEffect(() => {
   299→    load()
   300→  }, [load])
   301→
   302→  // realtime: refresh lastMessage + unread when a new message arrives
   303→  useEffect(() => {
   304→    if (!socket || !meId) return
   305→    const onNew = (raw: any) => {
   306→      if (!raw || raw.groupId) return
   307→      const peerId = raw.senderId === meId ? raw.receiverId : raw.senderId
   308→      if (!peerId) return
   309→      setConversations((prev) => {
   310→        const exists = prev.find((c) => c.id === peerId)
   311→        const lm = {
   312→          content: raw.content || '',
   313→          type: raw.type || 'text',
   314→          senderId: raw.senderId,
   315→          createdAt: toISO(raw.createdAt),
   316→          status: raw.status || 'SENT',
   317→        }
   318→        const unreadInc = raw.receiverId === meId ? 1 : 0
   319→        if (exists) {
   320→          const updated: Conversation = {
   321→            ...exists,
   322→            lastMessage: lm,
   323→            unread: exists.unread + unreadInc,
   324→          }
   325→          return [updated, ...prev.filter((c) => c.id !== peerId)]
   326→        }
   327→        // unknown peer — refetch to populate name/photo
   328→        load()
   329→        return prev
   330→      })
   331→    }
   332→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   333→      setConversations((prev) =>
   334→        prev.map((c) =>
   335→          c.lastMessage && c.lastMessage.senderId === meId && c.lastMessage.status !== status
   336→            ? { ...c, lastMessage: { ...c.lastMessage, status } }
   337→            : c
   338→        )
   339→      )
   340→    }
   341→    const onGroupNew = (raw: any) => {
   342→      if (!raw?.groupId) return
   343→      // bump group to top by re-fetching once (lightweight)
   344→      load()
   345→    }
   346→    socket.on(RT.MESSAGE_NEW, onNew)
   347→    socket.on(RT.MESSAGE_STATUS, onStatus as any)
   348→    socket.on(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   349→    return () => {
   350→      socket.off(RT.MESSAGE_NEW, onNew)
   351→      socket.off(RT.MESSAGE_STATUS, onStatus as any)
   352→      socket.off(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   353→    }
   354→  }, [socket, meId, load])
   355→
   356→  const q = query.trim().toLowerCase()
   357→  const filteredConvs = q
   358→    ? conversations.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   359→    : conversations
   360→  const filteredGroups = q
   361→    ? groups.filter((g) => g.name.toLowerCase().includes(q) || (g.description || '').toLowerCase().includes(q))
   362→    : groups
   363→
   364→  const totalUnread = conversations.reduce((s, c) => s + c.unread, 0)
   365→
   366→  return (
   367→    <div className="flex h-full flex-col">
   368→      <div className="border-b border-border px-4 py-3">
   369→        <div className="mb-3 flex items-center justify-between">
   370→          <h1 className="text-xl font-bold">Chats</h1>
   371→          {totalUnread > 0 && (
   372→            <span className="rounded-full bg-primary/15 px-2 py-0.5 text-xs font-semibold text-primary">
   373→              {totalUnread} unread
   374→            </span>
   375→          )}
   376→        </div>
   377→        <div className="relative">
   378→          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   379→          <Input
   380→            value={query}
   381→            onChange={(e) => setQuery(e.target.value)}
   382→            placeholder="Search chats and groups"
   383→            className="rounded-xl pl-9"
   384→          />
   385→        </div>
   386→      </div>
   387→
   388→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'chats' | 'groups')} className="flex flex-1 flex-col overflow-hidden">
   389→        <TabsList className="mx-4 mt-3 grid grid-cols-2 rounded-xl">
   390→          <TabsTrigger value="chats" className="text-xs">
   391→            Chats
   392→            {totalUnread > 0 && (
   393→              <span className="ml-1 grid h-4 min-w-4 place-items-center rounded-full bg-primary px-1 text-[9px] font-bold text-primary-foreground">
   394→                {totalUnread > 9 ? '9+' : totalUnread}
   395→              </span>
   396→            )}
   397→          </TabsTrigger>
   398→          <TabsTrigger value="groups" className="text-xs">
   399→            Groups
   400→            {groups.length > 0 && (
   401→              <span className="ml-1 text-[10px] font-normal text-muted-foreground">{groups.length}</span>
   402→            )}
   403→          </TabsTrigger>
   404→        </TabsList>
   405→
   406→        <TabsContent value="chats" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   407→          {loading ? (
   408→            <div className="space-y-1 p-2">
   409→              {[0, 1, 2, 3, 4].map((i) => (
   410→                <ConvRowSkeleton key={i} />
   411→              ))}
   412→            </div>
   413→          ) : filteredConvs.length === 0 ? (
   414→            <EmptyState
   415→              icon={<MessageCircle size={36} />}
   416→              title={q ? 'No matching chats' : 'No conversations yet'}
   417→              description={q ? 'Try a different search.' : 'Add a friend and say hello to start chatting.'}
   418→            />
   419→          ) : (
   420→            <div className="space-y-0.5">
   421→              {filteredConvs.map((c) => (
   422→                <ConversationRow
   423→                  key={c.id}
   424→                  conversation={c}
   425→                  online={isOnline(c.id)}
   426→                  meId={meId}
   427→                  onOpen={() => openChat(c.id)}
   428→                  onTogglePin={async () => {
   429→                    try {
   430→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, pinned: !c.pinned }) })
   431→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, pinned: !x.pinned } : x)))
   432→                    } catch (e: any) { toast.error(e.message) }
   433→                  }}
   434→                  onToggleMute={async () => {
   435→                    try {
   436→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, muted: !c.muted }) })
   437→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, muted: !x.muted } : x)))
   438→                    } catch (e: any) { toast.error(e.message) }
   439→                  }}
   440→                />
   441→              ))}
   442→            </div>
   443→          )}
   444→        </TabsContent>
   445→
   446→        <TabsContent value="groups" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   447→          {loading ? (
   448→            <div className="space-y-1 p-2">
   449→              {[0, 1, 2, 3].map((i) => (
   450→                <ConvRowSkeleton key={i} />
   451→              ))}
   452→            </div>
   453→          ) : filteredGroups.length === 0 ? (
   454→            <EmptyState
   455→              icon={<Users size={36} />}
   456→              title={q ? 'No matching groups' : 'No groups yet'}
   457→              description={q ? 'Try a different search.' : 'Create a group from the Groups tab to chat with multiple friends.'}
   458→            />
   459→          ) : (
   460→            <div className="space-y-0.5">
   461→              {filteredGroups.map((g) => (
   462→                <GroupRow key={g.id} group={g} onOpen={() => openGroupChat(g.id)} />
   463→              ))}
   464→            </div>
   465→          )}
   466→        </TabsContent>
   467→      </Tabs>
   468→    </div>
   469→  )
   470→}

   472→function ConversationRow({
   473→  conversation,
   474→  online,
   475→  meId,
   476→  onOpen,
   477→  onTogglePin,
   478→  onToggleMute,
   479→}: {
   480→  conversation: Conversation
   481→  online: boolean
   482→  meId?: string
   483→  onOpen: () => void
   484→  onTogglePin: () => void
   485→  onToggleMute: () => void
   486→}) {
   487→  const [menuOpen, setMenuOpen] = useState(false)
   488→  const pressTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   489→  const [pinned, setPinned] = useState(!!conversation.pinned)
   490→  const [muted, setMuted] = useState(!!conversation.muted)
   491→
   492→  useEffect(() => {
   493→    setPinned(!!conversation.pinned)
   494→    setMuted(!!conversation.muted)
   495→  }, [conversation.pinned, conversation.muted])
   496→
   497→  const startPress = () => {
   498→    pressTimer.current = setTimeout(() => setMenuOpen(true), 500)
   499→  }
   500→  const cancelPress = () => {
   501→    if (pressTimer.current) clearTimeout(pressTimer.current)
   502→  }
   503→
   504→  return (
   505→    <div className="relative">
   506→      <button
   507→        onClick={onOpen}
   508→        onPointerDown={startPress}
   509→        onPointerUp={cancelPress}
   510→        onPointerLeave={cancelPress}
   511→        className={cn(
   512→          'flex w-full items-center gap-3 rounded-xl p-2.5 text-left transition-colors hover:bg-accent/50',
   513→          pinned && 'bg-primary/5'
   514→        )}
   515→      >
   516→        <UserAvatar name={conversation.name} src={conversation.photoUrl} size={48} online={online} />
   517→        <div className="min-w-0 flex-1">
   518→          <div className="flex items-baseline justify-between gap-2">
   519→            <span className="flex min-w-0 items-center gap-1 truncate text-sm font-semibold">
   520→              <span className="truncate">{conversation.name}</span>
   521→              {pinned && <Pin size={11} className="shrink-0 text-primary" />}
   522→              {muted && <BellOff size={11} className="shrink-0 text-muted-foreground" />}
   523→            </span>
   524→            {conversation.lastMessage && (
   525→              <span className="shrink-0 text-[10px] text-muted-foreground">
   526→                {timeAgo(conversation.lastMessage.createdAt)}
   527→              </span>
   528→            )}
   529→          </div>
   530→          <div className="flex items-center justify-between gap-2">
   531→            <span
   532→              className={cn(
   533→                'truncate text-xs',
   534→                conversation.unread > 0 && !muted ? 'font-medium text-foreground' : 'text-muted-foreground'
   535→              )}
   536→            >
   537→              {lastMessagePreview(conversation, meId)}
   538→            </span>
   539→            {conversation.unread > 0 && (
   540→              <span
   541→                className={cn(
   542→                  'grid h-5 min-w-5 shrink-0 place-items-center rounded-full px-1.5 text-[10px] font-bold',
   543→                  muted ? 'bg-muted text-muted-foreground' : 'bg-primary text-primary-foreground'
   544→                )}
   545→              >
   546→                {conversation.unread > 9 ? '9+' : conversation.unread}
   547→              </span>
   548→            )}
   549→          </div>
   550→        </div>
   551→      </button>
   552→      {menuOpen && (
   553→        <>

   616→// ---------------- chat room (direct + group) ----------------
   617→function ChatRoom({
   618→  mode,
   619→  peerId,
   620→  onBack,
   621→}: {
   622→  mode: 'direct' | 'group'
   623→  peerId: string
   624→  onBack: () => void
   625→}) {
   626→  const { user: me } = useAuth()
   627→  const { socket, isOnline } = useSocket()
   628→  const [messages, setMessages] = useState<Message[]>([])
   629→  const [peerInfo, setPeerInfo] = useState<PeerInfo | null>(null)
   630→  const [loading, setLoading] = useState(true)
   631→  const [text, setText] = useState('')
   632→  const [peerTyping, setPeerTyping] = useState(false)
   633→  const [uploading, setUploading] = useState(false)
   634→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   635→  const [loadingOlder, setLoadingOlder] = useState(false)
   636→  const [hasMore, setHasMore] = useState(true)
   637→  const [searchQuery, setSearchQuery] = useState('')
   638→  const [searchOpen, setSearchOpen] = useState(false)
   639→  const [recording, setRecording] = useState(false)
   640→  const [recordSecs, setRecordSecs] = useState(0)
   641→  const [deleteTarget, setDeleteTarget] = useState<Message | null>(null)
   642→  const [forwardTarget, setForwardTarget] = useState<Message | null>(null)
   643→  const [settingsOpen, setSettingsOpen] = useState(false)
   644→  const [starredDialogOpen, setStarredDialogOpen] = useState(false)
   645→  const [disappearing, setDisappearing] = useState<'OFF' | '24H' | '7D'>('OFF')
   646→  const [wallpaper, setWallpaper] = useState<string | null>(null)
   647→
   648→  const scrollRef = useRef<HTMLDivElement>(null)
   649→  const bottomRef = useRef<HTMLDivElement>(null)
   650→  const fileInputRef = useRef<HTMLInputElement>(null)
   651→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   652→  const atBottomRef = useRef(true)
   653→  const oldestTsRef = useRef<string | null>(null)
   654→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   655→  const recordedChunksRef = useRef<Blob[]>([])
   656→  const recordStreamRef = useRef<MediaStream | null>(null)
   657→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   658→  const prevHeightRef = useRef(0)
   659→  const skipAutoScrollRef = useRef(false)
   660→  const meId = me?.id
   661→
   662→  // ---- load history + peer info ----
   663→  useEffect(() => {
   664→    let cancelled = false
   665→    setLoading(true)
   666→    setMessages([])
   667→    setHasMore(true)
   668→    setPeerTyping(false)
   669→    setPeerInfo(null)
   670→    oldestTsRef.current = null
   671→    atBottomRef.current = true
   672→
   673→    const endpoint = `/api/messages?${mode === 'direct' ? 'userId' : 'groupId'}=${encodeURIComponent(peerId)}`
   674→
   675→    Promise.all([
   676→      api<{ messages: Message[] }>(endpoint)
   677→        .then((d) => d.messages)
   678→        .catch((e: any) => {
   679→          toast.error(e?.message || 'Failed to load messages')
   680→          return [] as Message[]
   681→        }),
   682→      mode === 'direct'
   683→        ? api<{ friends: Friend[] }>('/api/friends')
   684→            .then((d) => d.friends.find((f) => f.id === peerId) || null)
   685→            .then((friend): PeerInfo | null => {
   686→              if (friend) {
   687→                return {
   688→                  name: friend.name,
   689→                  photoUrl: friend.photoUrl,
   690→                  username: friend.username,
   691→                  lastSeen: friend.lastSeen,
   692→                }
   693→              }
   694→              // fallback to conversations endpoint (peer may not be a friend yet but has messages)
   695→              return api<{ conversations: Conversation[] }>('/api/messages/conversations')
   696→                .then((d) => {
   697→                  const c = d.conversations.find((c) => c.id === peerId)
   698→                  return c
   699→                    ? { name: c.name, photoUrl: c.photoUrl, username: c.username, lastSeen: null }
   700→                    : null
   701→                })
   702→                .catch(() => null)
   703→            })
   704→            .catch(() => null)
   705→        : api<{ groups: GroupLite[] }>('/api/groups')
   706→            .then((d) => {
   707→              const g = d.groups.find((g) => g.id === peerId)
   708→              return g
   709→                ? { name: g.name, photoUrl: g.photoUrl, memberCount: g.memberCount }
   710→                : null
   711→            })
   712→            .catch(() => null),
   713→    ])
   714→      .then(([msgs, info]) => {
   715→        if (cancelled) return
   716→        const normalized = msgs.map(normalizeMessage)
   717→        setMessages(normalized)
   718→        if (normalized.length > 0) oldestTsRef.current = normalized[0].createdAt
   719→        if (normalized.length < 50) setHasMore(false)
   720→        if (info) setPeerInfo(info)
   721→      })
   722→      .finally(() => {
   723→        if (!cancelled) setLoading(false)
   724→      })
   725→
   726→    return () => {
   727→      cancelled = true
   728→    }
   729→  }, [mode, peerId])
   730→
   731→  // ---- load chat preferences (disappearing + wallpaper) for this peer ----
   732→  useEffect(() => {
   733→    setDisappearing('OFF')
   734→    setWallpaper(null)
   735→    let cancelled = false
   736→    api<{ preferences: Array<{ peerId: string; disappearing: string; wallpaper: string | null }> }>(
   737→      '/api/messages/preferences'
   738→    )
   739→      .then((d) => {
   740→        if (cancelled) return
   741→        const p = d.preferences?.find((x) => x.peerId === peerId)
   742→        if (p) {
   743→          setDisappearing(p.disappearing === '24H' || p.disappearing === '7D' ? p.disappearing : 'OFF')
   744→          setWallpaper(p.wallpaper ?? null)
   745→        }
   746→      })
   747→      .catch(() => {})
   748→    return () => {
   749→      cancelled = true
   750→    }
   751→  }, [peerId])
   752→
   753→  // ---- jump to bottom after first load ----
   754→  useEffect(() => {
   755→    if (loading) return
   756→    const id = requestAnimationFrame(() => {
   757→      bottomRef.current?.scrollIntoView({ block: 'end' })
   758→    })
   759→    return () => cancelAnimationFrame(id)
   760→  }, [loading])
   761→
   762→  // ---- auto-scroll on new messages ----
   763→  useEffect(() => {
   764→    if (loading) return
   765→    if (skipAutoScrollRef.current) {
   766→      skipAutoScrollRef.current = false
   767→      return
   768→    }
   769→    if (loadingOlder) return
   770→    if (!atBottomRef.current) return
   771→    const id = requestAnimationFrame(() => {
   772→      bottomRef.current?.scrollIntoView({ block: 'end', behavior: 'smooth' })
   773→    })
   774→    return () => cancelAnimationFrame(id)
   775→  }, [messages.length, peerTyping, loading, loadingOlder])
   776→
   777→  // ---- socket subscriptions ----
   778→  useEffect(() => {
   779→    if (!socket || !meId) return
   780→
   781→    const isRelevant = (raw: any): boolean => {
   782→      if (!raw) return false
   783→      if (mode === 'direct') {
   784→        if (raw.groupId) return false
   785→        return (
   786→          (raw.senderId === peerId && raw.receiverId === meId) ||
   787→          (raw.senderId === meId && raw.receiverId === peerId)
   788→        )
   789→      }
   790→      return raw.groupId === peerId
   791→    }
   792→
   793→    const onNew = (raw: any) => {
   794→      if (!isRelevant(raw)) return
   795→      const msg = normalizeMessage(raw)
   796→      setMessages((prev) => {
   797→        if (prev.some((m) => m.id === msg.id)) return prev
   798→        return [...prev, msg]
   799→      })
   800→      // mark realtime incoming 1:1 messages as seen
   801→      if (mode === 'direct' && msg.senderId === peerId && msg.receiverId === meId) {
   802→        socket.emit(RT.MESSAGE_SEEN, { messageId: msg.id })
   803→      }
   804→    }
   805→
   806→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   807→      setMessages((prev) =>
   808→        prev.map((m) => (m.id === id ? { ...m, status: status as MessageStatus } : m))
   809→      )
   810→    }
   811→
   812→    const onTyping = ({ userId, isTyping }: { userId: string; isTyping: boolean }) => {
   813→      if (mode === 'direct' && userId === peerId) setPeerTyping(isTyping)
   814→    }
   815→
   816→    const onReaction = ({ messageId, userId, emoji, action }: { messageId: string; userId: string; emoji: string; action: 'add' | 'remove' }) => {
   817→      setMessages((prev) =>
   818→        prev.map((m) => {
   819→          if (m.id !== messageId) return m
   820→          const reactions = m.reactions ? [...m.reactions] : []
   821→          if (action === 'add') {
   822→            if (!reactions.some((r) => r.userId === userId && r.emoji === emoji)) {
   823→              reactions.push({ emoji, userId, userName: '' })
   824→            }
   825→          } else {
   826→            const idx = reactions.findIndex((r) => r.userId === userId && r.emoji === emoji)
   827→            if (idx >= 0) reactions.splice(idx, 1)
   828→          }
   829→          return { ...m, reactions }
   830→        })
   831→      )
   832→    }
   833→
   834→    // Realtime message delete: when another participant deletes a message for
   835→    // everyone, the server emits message:deleted { messageId, scope:'everyone' }.
   836→    // We swap the local message for the "🚫 This message was deleted" placeholder.
   837→    const onDeleted = ({ messageId, scope }: { messageId: string; scope: string }) => {
   838→      if (scope !== 'everyone') return
   839→      setMessages((prev) =>
   840→        prev.map((m) =>
   841→          m.id === messageId
   842→            ? {
   843→                ...m,
   844→                deletedFor: 'everyone',
   845→                content: '',
   846→                fileUrl: null,
   847→                fileName: null,
   848→                fileSize: null,
   849→                mimeType: null,
   850→                type: 'text',
   851→                reactions: [],
   852→              }
   853→            : m
   854→        )
   855→      )
   856→    }
   857→
   858→    socket.on(RT.MESSAGE_NEW, onNew)
   859→    socket.on(RT.GROUP_MESSAGE_NEW, onNew)
   860→    socket.on(RT.MESSAGE_STATUS, onStatus)
   861→    socket.on(RT.TYPING_INDICATOR, onTyping)
   862→    socket.on(RT.REACTION_UPDATE, onReaction)
   863→    socket.on(RT.MESSAGE_DELETED, onDeleted)
   864→
   865→    return () => {
   866→      socket.off(RT.MESSAGE_NEW, onNew)
   867→      socket.off(RT.GROUP_MESSAGE_NEW, onNew)
   868→      socket.off(RT.MESSAGE_STATUS, onStatus)
   869→      socket.off(RT.TYPING_INDICATOR, onTyping)
   870→      socket.off(RT.REACTION_UPDATE, onReaction)
   871→      socket.off(RT.MESSAGE_DELETED, onDeleted)
   872→      if (mode === 'direct') {
   873→        socket.emit(RT.TYPING, { receiverId: peerId, isTyping: false })
   874→      }
   875→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current)
   876→    }
   877→  }, [socket, meId, peerId, mode])
   878→
   879→  // ---- toggle reaction on a message ----
   880→  const toggleReaction = useCallback(
   881→    (messageId: string, emoji: string) => {
   882→      if (!socket) return
   883→      setMessages((prev) =>
   884→        prev.map((m) => {
   885→          if (m.id !== messageId) return m
   886→          const reactions = m.reactions ? [...m.reactions] : []
   887→          const existing = reactions.find((r) => r.userId === meId && r.emoji === emoji)
   888→          if (existing) {
   889→            const idx = reactions.indexOf(existing)
   890→            reactions.splice(idx, 1)
   891→            socket.emit(RT.REACTION_REMOVE, { messageId, emoji })
   892→          } else {
   893→            reactions.push({ emoji, userId: meId, userName: me?.name || '' })
   894→            socket.emit(RT.REACTION_ADD, { messageId, emoji })
   895→          }
   896→          return { ...m, reactions }
   897→        })
   898→      )
   899→    },
   900→    [socket, meId, me]
   901→  )
   902→
   903→  // ---- reply (inserts a simple @reply prefix into the input) ----
   904→  const handleReply = useCallback((msg: Message) => {
   905→    setText((t) => (t.startsWith('@reply') ? t : `@reply ${t}`))
   906→  }, [])
   907→
   908→  // ---- delete message (for me / for everyone) ----
   909→  const confirmDelete = useCallback(
   910→    async (msg: Message, scope: 'me' | 'everyone') => {
   911→      setDeleteTarget(null)
   912→      try {
   913→        await api(`/api/messages/${msg.id}/delete`, {
   914→          method: 'POST',
   915→          body: JSON.stringify({ scope }),
   916→        })
   917→        if (scope === 'me') {
   918→          // optimistic: remove from local view immediately
   919→          setMessages((prev) => prev.filter((m) => m.id !== msg.id))
   920→          toast.success('Message deleted for you')
   921→        } else {
   922→          // tell other participants via realtime + swap local for placeholder
   923→          if (socket) {
   924→            socket.emit(RT.MESSAGE_DELETE, { messageId: msg.id, scope: 'everyone' })
   925→          }
   926→          setMessages((prev) =>
   927→            prev.map((m) =>
   928→              m.id === msg.id
   929→                ? {
   930→                    ...m,
   931→                    deletedFor: 'everyone',
   932→                    content: '',
   933→                    fileUrl: null,
   934→                    fileName: null,
   935→                    fileSize: null,
   936→                    mimeType: null,
   937→                    type: 'text',
   938→                    reactions: [],
   939→                  }
   940→                : m
   941→            )
   942→          )
   943→          toast.success('Message deleted for everyone')
   944→        }
   945→      } catch (e: any) {
   946→        toast.error(e?.message || 'Failed to delete message')
   947→      }
   948→    },
   949→    [socket]
   950→  )
   951→
   952→  // ---- toggle star on a message (optimistic + API) ----
   953→  const toggleStar = useCallback(async (msg: Message) => {
   954→    const next = !msg.starred
   955→    setMessages((prev) => prev.map((m) => (m.id === msg.id ? { ...m, starred: next } : m)))
   956→    try {
   957→      if (next) {
   958→        await api(`/api/messages/${msg.id}/star`, { method: 'POST' })
   959→        toast.success('Starred')
   960→      } else {
   961→        await api(`/api/messages/${msg.id}/star`, { method: 'DELETE' })
   962→        toast.success('Unstarred')
   963→      }
   964→    } catch (e: any) {
   965→      // revert on failure
   966→      setMessages((prev) => prev.map((m) => (m.id === msg.id ? { ...m, starred: !next } : m)))
   967→      toast.error(e?.message || 'Failed to update star')
   968→    }
   969→  }, [])
   970→
   971→  // ---- update disappearing messages preference ----
   972→  const handleDisappearingChange = useCallback(
   973→    async (value: 'OFF' | '24H' | '7D') => {
   974→      const prev = disappearing
   975→      setDisappearing(value)
   976→      try {
   977→        await api('/api/messages/preferences', {
   978→          method: 'POST',
   979→          body: JSON.stringify({ peerId, disappearing: value }),
   980→        })
   981→        toast.success(
   982→          value === 'OFF'
   983→            ? 'Disappearing messages off'
   984→            : `Messages disappear after ${value === '24H' ? '24 hours' : '7 days'}`
   985→        )
   986→      } catch (e: any) {
   987→        setDisappearing(prev)
   988→        toast.error(e?.message || 'Failed to update')
   989→      }
   990→    },
   991→    [peerId, disappearing]
   992→  )
   993→
   994→  // ---- update wallpaper preference ----
   995→  const handleWallpaperChange = useCallback(
   996→    async (value: string) => {
   997→      const prev = wallpaper
   998→      setWallpaper(value || null)
   999→      try {
  1000→        await api('/api/messages/preferences', {
  1001→          method: 'POST',
  1002→          body: JSON.stringify({ peerId, wallpaper: value || null }),
  1003→        })
  1004→        toast.success(value ? 'Wallpaper updated' : 'Wallpaper cleared')
  1005→      } catch (e: any) {
  1006→        setWallpaper(prev)
  1007→        toast.error(e?.message || 'Failed to update wallpaper')
  1008→      }
  1009→    },
  1010→    [peerId, wallpaper]
  1011→  )
  1012→
  1013→  // ---- scroll handling + infinite scroll up ----
  1014→  const loadOlder = useCallback(async () => {
  1015→    if (!oldestTsRef.current) return
  1016→    setLoadingOlder(true)
  1017→    const el = scrollRef.current
  1018→    prevHeightRef.current = el?.scrollHeight || 0
  1019→    try {
  1020→      const endpoint = `/api/messages?${mode === 'direct' ? 'userId' : 'groupId'}=${encodeURIComponent(peerId)}&before=${encodeURIComponent(oldestTsRef.current)}`
  1021→      const data = await api<{ messages: Message[] }>(endpoint)
  1022→      if (data.messages.length === 0) {
  1023→        setHasMore(false)
  1024→      } else {
  1025→        const normalized = data.messages.map(normalizeMessage)
  1026→        skipAutoScrollRef.current = true
  1027→        setMessages((prev) => [...normalized, ...prev])
  1028→        oldestTsRef.current = normalized[0]?.createdAt || oldestTsRef.current
  1029→        if (normalized.length < 50) setHasMore(false)
  1030→        // restore scroll position after DOM update
  1031→        requestAnimationFrame(() => {
  1032→          if (el) {
  1033→            const newHeight = el.scrollHeight
  1034→            el.scrollTop = newHeight - prevHeightRef.current
  1035→          }
  1036→        })
  1037→      }
  1038→    } catch (e: any) {
  1039→      toast.error(e?.message || 'Failed to load older messages')
  1040→    } finally {
  1041→      setLoadingOlder(false)
  1042→    }
  1043→  }, [mode, peerId])
  1044→
  1045→  const handleScroll = useCallback(() => {
  1046→    const el = scrollRef.current
  1047→    if (!el) return
  1048→    const distFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
  1049→    atBottomRef.current = distFromBottom < 80
  1050→    if (el.scrollTop < 60 && hasMore && !loadingOlder && !loading && oldestTsRef.current) {
  1051→      loadOlder()
  1052→    }
  1053→  }, [hasMore, loadingOlder, loading, loadOlder])
  1054→
  1055→  // ---- send text ----
  1056→  const sendMessage = useCallback(
  1057→    (partial: Pick<Message, 'content' | 'type' | 'fileUrl' | 'fileName' | 'fileSize' | 'mimeType'>) => {
  1058→      if (!socket || !meId) return
  1059→      const tempId = newTempId()
  1060→      const optimistic: Message = {
  1061→        id: tempId,
  1062→        tempId,
  1063→        senderId: meId,
  1064→        senderName: me?.name || null,
  1065→        senderPhoto: me?.profile?.photoUrl ?? null,
  1066→        receiverId: mode === 'direct' ? peerId : null,
  1067→        groupId: mode === 'group' ? peerId : null,
  1068→        content: partial.content,
  1069→        type: partial.type,
  1070→        fileUrl: partial.fileUrl ?? null,
  1071→        fileName: partial.fileName ?? null,
  1072→        fileSize: partial.fileSize ?? null,
  1073→        mimeType: partial.mimeType ?? null,
  1074→        status: 'SENDING',
  1075→        createdAt: new Date().toISOString(),
  1076→      }
  1077→      setMessages((prev) => [...prev, optimistic])
  1078→      atBottomRef.current = true
  1079→
  1080→      const payload: any = {
  1081→        tempId,
  1082→        content: partial.content,
  1083→        type: partial.type,

  1098→          }

  1103→            prev.map((m) => (m.tempId === tempId ? { ...m, status: 'SENT' } : m))
  1104→          )
