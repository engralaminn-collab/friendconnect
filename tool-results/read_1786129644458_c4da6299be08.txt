    73→    73→/**
    74→    74→ * Full-screen story viewer modal.
    75→    75→ * - Progress bars at top (one per story, active fills over ~5s)
    76→    76→ * - Auto-advance to next story after 5s; after last, close
    77→    77→ * - Tap left/right thirds to go prev/next, center third to close

    82→    82→ * - Keyboard: Escape to close, ArrowLeft/ArrowRight to navigate
    83→    83→ */
    84→    84→export function StoryViewer({ stories, user, onClose, startIndex, onHighlightChange }: StoryViewerProps) {
    85→    85→  const [index, setIndex] = useState(() => {
    86→    86→    if (typeof startIndex === 'number' && startIndex >= 0 && startIndex < stories.length) {
    87→    87→      return startIndex
    88→    88→    }
    89→    89→    return 0
    90→    90→  })
    91→    91→  const [imgError, setImgError] = useState(false)
    92→    92→  const [replyText, setReplyText] = useState('')
    93→    93→  const [sending, setSending] = useState(false)
    94→    94→  const [replies, setReplies] = useState<StoryReply[]>([])
    95→    95→  const [repliesOpen, setRepliesOpen] = useState(false)
    96→    96→  const [loadingReplies, setLoadingReplies] = useState(false)
    97→    97→  const [paused, setPaused] = useState(false)
    98→    98→  // Local override map for the highlight flag — keeps the icon in sync immediately
    99→    99→  // after a toggle, before the parent refetches the stories list.
   100→   100→  const [highlightOverrides, setHighlightOverrides] = useState<Record<string, boolean>>({})
   101→   101→  const [highlightBusy, setHighlightBusy] = useState(false)
   102→   102→  // Round 8: story reactions. `myReaction` is a local override for the
   103→   103→  // active story's my-reaction emoji — set optimistically on tap and
   104→   104→  // reconciled when the active story changes (via the effect below).
   105→   105→  // `reactionList` is the fresh list of reactions (with avatars + emoji)
   106→   106→  // fetched for the user's OWN story so they can see who reacted.
   107→   107→  const [myReaction, setMyReaction] = useState<string | null>(null)
   108→   108→  const [reactionList, setReactionList] = useState<StoryReaction[]>([])
   109→   109→  const [reactionsOpen, setReactionsOpen] = useState(false)
   110→   110→  const [loadingReactions, setLoadingReactions] = useState(false)
   111→   111→  const [reacting, setReacting] = useState(false)
   112→   112→  const viewedRef = useRef<Set<string>>(new Set())
   113→   113→  const touchStartX = useRef<number | null>(null)
   114→   114→  const progressBarRef = useRef<HTMLDivElement | null>(null)
   115→   115→  const rafRef = useRef<number | null>(null)
   116→   116→  const startRef = useRef<number>(0)
   117→   117→  const pausedAtRef = useRef<number | null>(null)
   118→   118→  const pausedAccRef = useRef<number>(0)
   119→   119→  const pausedRef = useRef(false)
   120→   120→  const goNextRef = useRef<() => void>(() => {})
   121→   121→
   122→   122→  const goNext = useCallback(() => {
   123→   123→    setIndex((i) => {
   124→   124→      if (i + 1 >= stories.length) {
   125→   125→        onClose()
   126→   126→        return i
   127→   127→      }
   128→   128→      return i + 1
   129→   129→    })
   130→   130→  }, [stories.length, onClose])
   131→   131→
   132→   132→  const goPrev = useCallback(() => {
   133→   133→    setIndex((i) => Math.max(0, i - 1))
   134→   134→  }, [])
   135→   135→
   136→   136→  // Reset image error state when index changes
   137→   137→  useEffect(() => {
   138→   138→    setImgError(false)
   139→   139→  }, [index])
   140→   140→
   141→   141→  // Reset reply text + close replies popover when story changes
   142→   142→  useEffect(() => {
   143→   143→    setReplyText('')
   144→   144→    setRepliesOpen(false)
   145→   145→  }, [index])
   146→   146→
   147→   147→  // Mark the active story as viewed (once per story id)
   148→   148→  useEffect(() => {
   149→   149→    const story = stories[index]
   150→   150→    if (!story) return
   151→   151→    if (viewedRef.current.has(story.id)) return
   152→   152→    viewedRef.current.add(story.id)
   153→   153→    api(`/api/stories/${story.id}/view`, { method: 'POST' }).catch(() => {
   154→   154→      /* silent — view tracking is best-effort */
   155→   155→    })
   156→   156→  }, [index, stories])
   157→   157→
   158→   158→  // Keep refs in sync (used by the rAF auto-advance timer)
   159→   159→  useEffect(() => {
   160→   160→    goNextRef.current = goNext
   161→   161→  }, [goNext])
   162→   162→  useEffect(() => {
   163→   163→    pausedRef.current = paused
   164→   164→  }, [paused])
   165→   165→
   166→   166→  // Fetch replies when viewing own story (and when story changes)
   167→   167→  useEffect(() => {
   168→   168→    if (!user.isMe) {
   169→   169→      setReplies([])
   170→   170→      return
   171→   171→    }
   172→   172→    const story = stories[index]
   173→   173→    if (!story) return
   174→   174→    let cancelled = false
   175→   175→    setLoadingReplies(true)
   176→   176→    api<{ replies: StoryReply[] }>(`/api/stories/${story.id}/replies`)
   177→   177→      .then((data) => {
   178→   178→        if (!cancelled) setReplies(data.replies)
   179→   179→      })
   180→   180→      .catch(() => {
   181→   181→        if (!cancelled) setReplies([])
   182→   182→      })
   183→   183→      .finally(() => {
   184→   184→        if (!cancelled) setLoadingReplies(false)
   185→   185→      })
   186→   186→    return () => {
   187→   187→      cancelled = true
   188→   188→    }
   189→   189→  }, [index, user.isMe, stories])
   190→   190→
   191→   191→  // Round 8: reconcile the local `myReaction` override with the active
   192→   192→  // story's `myReaction` from the API whenever the active story changes.
   193→   193→  // The override is set optimistically on tap and is otherwise synced from
   194→   194→  // the parent's stories prop (which is the source of truth from GET /api/stories).
   195→   195→  useEffect(() => {
   196→   196→    const story = stories[index]
   197→   197→    if (!story) return
   198→   198→    setMyReaction(story.myReaction ?? null)
   199→   199→    setReactionsOpen(false)
   200→   200→  }, [index, stories])
   201→   201→
   202→   202→  // Round 8: fetch the full reactions list (with avatars + emoji) for the
   203→   203→  // user's OWN story so they can see who reacted. Mirrors the replies fetch
   204→   204→  // above. For other users' stories we don't fetch (their reactions are
   205→   205→  // private — only the count is shown in the header pill).
   206→   206→  useEffect(() => {
   207→   207→    if (!user.isMe) {
   208→   208→      setReactionList([])
   209→   209→      return
   210→   210→    }
   211→   211→    const story = stories[index]
   212→   212→    if (!story) return
   213→   213→    let cancelled = false
   214→   214→    setLoadingReactions(true)
   215→   215→    api<{ reactions: StoryReaction[] }>(`/api/stories/${story.id}/reactions`)
   216→   216→      .then((data) => {
   217→   217→        if (!cancelled) setReactionList(data.reactions || [])
   218→   218→      })
   219→   219→      .catch(() => {
   220→   220→        if (!cancelled) setReactionList([])
   221→   221→      })
   222→   222→      .finally(() => {
   223→   223→        if (!cancelled) setLoadingReactions(false)
   224→   224→      })
   225→   225→    return () => {
   226→   226→      cancelled = true
   227→   227→    }
   228→   228→  }, [index, user.isMe, stories])
   229→   229→
   230→   230→  // Auto-advance timer using requestAnimationFrame (pauses while the reply input is focused).
   231→   231→  // Replaces the previous framer-motion progress bar so we can cleanly pause/resume on input focus.
   232→   232→  useEffect(() => {
   233→   233→    startRef.current = Date.now()
   234→   234→    pausedAtRef.current = pausedRef.current ? Date.now() : null
   235→   235→    pausedAccRef.current = 0
   236→   236→    if (progressBarRef.current) progressBarRef.current.style.width = '0%'
   237→   237→
   238→   238→    const tick = () => {
   239→   239→      if (pausedAtRef.current === null) {
   240→   240→        const elapsed = Date.now() - startRef.current - pausedAccRef.current
   241→   241→        const p = Math.min(1, elapsed / STORY_DURATION_MS)
   242→   242→        if (progressBarRef.current) {
   243→   243→          progressBarRef.current.style.width = `${p * 100}%`
   244→   244→        }
   245→   245→        if (p >= 1) {
   246→   246→          goNextRef.current()
   247→   247→          return
   248→   248→        }
   249→   249→      }
   250→   250→      rafRef.current = requestAnimationFrame(tick)
   251→   251→    }
   252→   252→    rafRef.current = requestAnimationFrame(tick)
   253→   253→
   254→   254→    return () => {
   255→   255→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   256→   256→    }
   257→   257→  }, [index])
   258→   258→
   259→   259→  // Pause/resume handler (driven by reply input focus)
   260→   260→  useEffect(() => {
   261→   261→    if (paused) {
   262→   262→      pausedAtRef.current = Date.now()
   263→   263→    } else if (pausedAtRef.current !== null) {
   264→   264→      pausedAccRef.current += Date.now() - pausedAtRef.current
   265→   265→      pausedAtRef.current = null
   266→   266→    }
   267→   267→  }, [paused])
   268→   268→
   269→   269→  // Lock body scroll while open
   270→   270→  useEffect(() => {
   271→   271→    const prev = document.body.style.overflow
   272→   272→    document.body.style.overflow = 'hidden'
   273→   273→    return () => {
   274→   274→      document.body.style.overflow = prev
   275→   275→    }
   276→   276→  }, [])
   277→   277→
   278→   278→  // Keyboard navigation
   279→   279→  useEffect(() => {
   280→   280→    const onKey = (e: KeyboardEvent) => {
   281→   281→      if (e.key === 'Escape') {
   282→   282→        e.preventDefault()
   283→   283→        onClose()
   284→   284→      } else if (e.key === 'ArrowRight') {
   285→   285→        e.preventDefault()
   286→   286→        goNext()
   287→   287→      } else if (e.key === 'ArrowLeft') {
   288→   288→        e.preventDefault()
   289→   289→        goPrev()
   290→   290→      }
   291→   291→    }
   292→   292→    window.addEventListener('keydown', onKey)
   293→   293→    return () => window.removeEventListener('keydown', onKey)
   294→   294→  }, [onClose, goNext, goPrev])
   295→   295→
   296→   296→  // Touch swipe support (left = next, right = prev, down = close)
   297→   297→  const onTouchStart = (e: React.TouchEvent) => {
   298→   298→    touchStartX.current = e.touches[0]?.clientX ?? null
   299→   299→  }
   300→   300→  const onTouchEnd = (e: React.TouchEvent) => {
   301→   301→    if (touchStartX.current == null) return
   302→   302→    const dx = (e.changedTouches[0]?.clientX ?? touchStartX.current) - touchStartX.current
   303→   303→    touchStartX.current = null
   304→   304→    if (Math.abs(dx) < 50) return
   305→   305→    if (dx < 0) goNext()
   306→   306→    else goPrev()
   307→   307→  }
   308→   308→
   309→   309→  // Post a reply to the current story (friends only — backend enforces).
   310→   310→  const handleSendReply = useCallback(async () => {
   311→   311→    const story = stories[index]
   312→   312→    if (!story) return
   313→   313→    const content = replyText.trim()
   314→   314→    if (!content || sending) return
   315→   315→    setSending(true)
   316→   316→    try {
   317→   317→      const data = await api<{ reply: StoryReply }>(`/api/stories/${story.id}/replies`, {
   318→   318→        method: 'POST',
   319→   319→        body: JSON.stringify({ content }),
   320→   320→      })
   321→   321→      // If we happen to be viewing our own story, surface our own reply in the list.
   322→   322→      if (user.isMe) {
   323→   323→        setReplies((prev) => [...prev, data.reply])
   324→   324→      }
   325→   325→      setReplyText('')
   326→   326→      toast.success('Reply sent')
   327→   327→    } catch (e: any) {
   328→   328→      toast.error(e?.message || 'Failed to send reply')
   329→   329→    } finally {
   330→   330→      setSending(false)
   331→   331→    }
   332→   332→  }, [stories, index, replyText, sending, user.isMe])
   333→   333→
   334→   334→  // Toggle highlight on the current story (owner only). Optimistic local update
   335→   335→  // so the bookmark icon flips immediately; the parent is notified so it can
   336→   336→  // refresh its stories list (the highlighted story moves between the
   337→   337→  // "stories" and "highlights" buckets on the next fetch).
   338→   338→  const handleToggleHighlight = useCallback(async () => {
   339→   339→    const story = stories[index]
   340→   340→    if (!story || highlightBusy) return
   341→   341→    const next = !(highlightOverrides[story.id] ?? story.isHighlight)
   342→   342→    setHighlightOverrides((prev) => ({ ...prev, [story.id]: next }))
   343→   343→    setHighlightBusy(true)
   344→   344→    try {
   345→   345→      await api(`/api/stories/${story.id}/highlight`, {
   346→   346→        method: 'POST',
   347→   347→        body: JSON.stringify({ highlight: next }),
   348→   348→      })
   349→   349→      toast.success(next ? 'Added to highlights' : 'Removed from highlights')
   350→   350→      onHighlightChange?.(story.id, next)
   351→   351→    } catch (e: any) {
   352→   352→      // Revert the optimistic update.
   353→   353→      setHighlightOverrides((prev) => ({ ...prev, [story.id]: !next }))
   354→   354→      toast.error(e?.message || 'Failed to toggle highlight')
   355→   355→    } finally {
   356→   356→      setHighlightBusy(false)
   357→   357→    }
   358→   358→  }, [stories, index, highlightOverrides, highlightBusy, onHighlightChange])
   359→   359→
   360→   360→  // Round 8: react to the current story with an emoji. The backend upserts
   361→   361→  // (one reaction per user per story — replaces any existing reaction by me).
   362→   362→  // We update the local `myReaction` state optimistically so the active emoji
   363→   363→  // flips immediately; on error we revert. Tapping the same emoji again is a
   364→   364→  // no-op visually but still POSTs (the backend re-creates the same reaction —
   365→   365→  // harmless + idempotent). Doesn't pause the auto-advance timer (reactions
   366→   366→  // are quick taps, the user has 5s to react before the story advances).
   367→   367→  const handleReact = useCallback(async (emoji: string) => {
   368→   368→    const story = stories[index]
   369→   369→    if (!story || reacting) return
   370→   370→    const prev = myReaction
   371→   371→    setMyReaction(emoji)
   372→   372→    setReacting(true)
   373→   373→    try {
   374→   374→      await api(`/api/stories/${story.id}/reactions`, {
   375→   375→        method: 'POST',
   376→   376→        body: JSON.stringify({ emoji }),
   377→   377→      })
   378→   378→      toast.success('Reacted')
   379→   379→      // If we're viewing our own story, surface our own reaction in the
   380→   380→      // popover list. The fresh GET /api/stories/[id]/reactions effect will
   381→   381→      // reconcile on the next story change anyway, but this keeps the list
   382→   382→      // in sync immediately so the count + popover are accurate right now.
   383→   383→      if (user.isMe) {
   384→   384→        setReactionList((prevList) => {
   385→   385→          // Remove any existing reaction by me, then add the new one.
   386→   386→          const filtered = prevList.filter((r) => r.userId !== user.userId)
   387→   387→          return [
   388→   388→            ...filtered,
   389→   389→            {
   390→   390→              id: `local-${story.id}-${user.userId}`,
   391→   391→              userId: user.userId,
   392→   392→              userName: user.name,
   393→   393→              userPhoto: user.photoUrl,
   394→   394→              emoji,
   395→   395→              createdAt: new Date().toISOString(),
   396→   396→            },
   397→   397→          ]
   398→   398→        })
   399→   399→      }
   400→   400→    } catch (e: any) {
   401→   401→      setMyReaction(prev)
   402→   402→      toast.error(e?.message || 'Failed to react')
   403→   403→    } finally {
   404→   404→      setReacting(false)
   405→   405→    }
   406→   406→  }, [stories, index, reacting, myReaction, user.isMe, user.userId, user.name, user.photoUrl])
   407→   407→
   408→   408→  if (!stories.length) {
   409→   409→    return null
   410→   410→  }
   411→   411→
   412→   412→  const current = stories[index]
   413→   413→  if (!current) return null
   414→   414→
   415→   415→  const bgStyle = current.bg || 'linear-gradient(135deg, var(--color-primary), var(--color-accent))'
   416→   416→  const isHighlighted = highlightOverrides[current.id] ?? current.isHighlight
   417→   417→
   418→   418→  // Round 8: derived reaction count for the header pill. For own stories we
   419→   419→  // prefer the fresh reactions-list length (the GET /api/stories/[id]/reactions
   420→   420→  // fetch); for other users' stories we adjust the API count for the local
   421→   421→  // myReaction override so the pill flips to N+1 immediately on first react.
   422→   422→  // Declared here (BEFORE the JSX) to avoid a temporal-dead-zone warning.
   423→   423→  const headerReactionCount = user.isMe
   424→   424→    ? Math.max(reactionList.length, current.reactionCount || 0)
   425→   425→    : (current.reactionCount || 0) + (myReaction && !current.myReaction ? 1 : 0)
   426→   426→
   427→   427→  return (
   428→   428→    <motion.div
   429→   429→      initial={{ opacity: 0 }}
   430→   430→      animate={{ opacity: 1 }}
   431→   431→      exit={{ opacity: 0 }}
   432→   432→      transition={{ duration: 0.2 }}
   433→   433→      className="fixed inset-0 z-[90] bg-black select-none"
   434→   434→      onTouchStart={onTouchStart}
   435→   435→      onTouchEnd={onTouchEnd}
   436→   436→    >
   437→   437→      {/* Story content (crossfades between stories) */}
   438→   438→      <AnimatePresence>
   439→   439→        <motion.div
   440→   440→          key={current.id}
   441→   441→          initial={{ opacity: 0 }}
   442→   442→          animate={{ opacity: 1 }}
   443→   443→          exit={{ opacity: 0 }}
   444→   444→          transition={{ duration: 0.25 }}
   445→   445→          className="absolute inset-0"
   446→   446→        >
   447→   447→          {current.type === 'image' && !imgError ? (
   448→   448→            <img
   449→   449→              src={current.content}
   450→   450→              alt={user.name ? `${user.name}'s story` : 'Story'}
   451→   451→              className="h-full w-full object-contain"
   452→   452→              onError={() => setImgError(true)}
   453→   453→            />
