   397→function PhotoFilterPicker({
   398→  user,
   399→  refresh,
   400→}: {
   401→  user: AuthUser
   402→  refresh: () => Promise<AuthUser | null>
   403→}) {
   404→  const [open, setOpen] = useState(false)
   405→  const [applying, setApplying] = useState<string | null>(null)
   406→  const profile = user.profile
   407→  const current = profile?.photoFilter || 'none'
   408→
   409→  const apply = async (value: string) => {
   410→    if (applying) return
   411→    setApplying(value)
   412→    try {
   413→      // Backend treats empty string as null (clears the filter).
   414→      const body = value === 'none' ? { photoFilter: '' } : { photoFilter: value }
   415→      await api('/api/users/me', { method: 'PUT', body: JSON.stringify(body) })
   416→      await refresh()
   417→      toast.success('Photo filter updated')
   418→      setOpen(false)
   419→    } catch (e: any) {
   420→      toast.error(e.message || 'Failed to update filter')
   421→    } finally {
   422→      setApplying(null)
   423→    }
   424→  }
   425→
   426→  return (
   427→    <Popover open={open} onOpenChange={setOpen}>
   428→      <PopoverTrigger asChild>
   429→        <button
   430→          type="button"
   431→          className="grid h-8 w-8 place-items-center rounded-full bg-primary/15 text-primary shadow-sm transition hover:bg-primary/25"
   432→          aria-label="Photo filters"
   433→          title="Photo filters"
   434→        >
   435→          <Sparkles size={14} />
   436→        </button>
   437→      </PopoverTrigger>
   438→      <PopoverContent className="w-72 p-3" align="end" sideOffset={8}>
   439→        <div className="mb-2 flex items-center gap-1.5">
   440→          <Sparkles size={14} className="text-primary" />
   441→          <span className="text-sm font-semibold">Photo filter</span>
   442→        </div>
   443→        <div className="grid grid-cols-3 gap-2">
   444→          {PHOTO_FILTERS.map((f) => {
   445→            const selected = current === f.value
   446→            const isApplying = applying === f.value
   447→            return (
   448→              <button
   449→                key={f.id}
   450→                type="button"
   451→                onClick={() => apply(f.value)}
   452→                disabled={!!applying}
   453→                className={cn(
   454→                  'relative flex flex-col items-center gap-1 rounded-xl p-2 transition hover:bg-accent',
   455→                  selected && 'ring-2 ring-primary'
   456→                )}

   458→                <UserAvatar
   459→                  name={user.name}
   460→                  src={profile?.photoUrl}
   461→                  size={48}
   462→                  filter={f.value === 'none' ? undefined : f.value}
   463→                />
   464→                <span className="text-[10px] font-medium text-muted-foreground">{f.name}</span>
   465→                {isApplying && (
   466→                  <span className="absolute inset-0 grid place-items-center rounded-xl bg-background/60">

   480→/**
   481→ * Round 9: PhotoCropDialog — replaces the old "upload immediately" flow with
   482→ * a client-side crop step. The parent picks an image and hands us its data
   483→ * URL; we render it in a square viewport with a dark mask outside the crop
   484→ * region, a 1×–3× zoom slider, and drag-to-pan. On Apply we draw the visible
   485→ * square region to a 256×256 canvas, convert it to a JPEG Blob, upload it
   486→ * via `uploadFile`, PUT the new URL to /api/users/me, and refresh the
   487→ * session. The crop step is mandatory (no skip) so every avatar is square.
   488→ */
   489→const CROP_CONTAINER_SIZE = 280
   490→const CROP_OUTPUT_SIZE = 256
   491→
   492→function PhotoCropDialog({
   493→  src,
   494→  onClose,
   495→  refresh,
   496→}: {
   497→  src: string
   498→  onClose: () => void
   499→  refresh: () => Promise<AuthUser | null>
   500→}) {
   501→  const imgRef = useRef<HTMLImageElement | null>(null)
   502→  const [loaded, setLoaded] = useState(false)
   503→  const [imgSize, setImgSize] = useState<{ nw: number; nh: number } | null>(null)
   504→  const [zoom, setZoom] = useState(1) // 1.0 – 3.0
   505→  const [pan, setPan] = useState<{ x: number; y: number }>({ x: 0, y: 0 })
   506→  const [applying, setApplying] = useState(false)
   507→  // Pointer-drag state. Stored in a ref so move/up handlers can read the
   508→  // latest values without re-creating the handlers each render.
   509→  const dragRef = useRef<{ startX: number; startY: number; panX: number; panY: number } | null>(null)
   510→
   511→  // Reset transient state when a new image is handed in.
   512→  useEffect(() => {
   513→    setLoaded(false)
   514→    setImgSize(null)
   515→    setZoom(1)
   516→    setPan({ x: 0, y: 0 })
   517→  }, [src])
   518→
   519→  const onImgLoad = () => {
   520→    const img = imgRef.current
   521→    if (!img) return
   522→    setImgSize({ nw: img.naturalWidth, nh: img.naturalHeight })
   523→    setLoaded(true)
   524→  }
   525→  const onImgError = () => {
   526→    toast.error('Could not load image')
   527→    onClose()
   528→  }
   529→
   530→  // fillScale = the scale at which the image just fills the container in its
   531→  // smaller dimension (object-cover semantics). At zoom=1 the visible region
   532→  // is the centered square of size min(nw,nh) in natural coords.
   533→  const fillScale = imgSize
   534→    ? CROP_CONTAINER_SIZE / Math.min(imgSize.nw, imgSize.nh)
   535→    : 1
   536→  const effScale = fillScale * zoom
   537→  const scaledW = imgSize ? imgSize.nw * effScale : 0
   538→  const scaledH = imgSize ? imgSize.nh * effScale : 0
   539→  const maxPanX = Math.max(0, (scaledW - CROP_CONTAINER_SIZE) / 2)
   540→  const maxPanY = Math.max(0, (scaledH - CROP_CONTAINER_SIZE) / 2)
   541→  const clampPan = (p: { x: number; y: number }) => ({
   542→    x: Math.max(-maxPanX, Math.min(maxPanX, p.x)),
   543→    y: Math.max(-maxPanY, Math.min(maxPanY, p.y)),
   544→  })
   545→  const clampedPan = clampPan(pan)
   546→
   547→  const onPointerDown = (e: React.PointerEvent) => {
   548→    if (!loaded) return
   549→    e.currentTarget.setPointerCapture(e.pointerId)
   550→    dragRef.current = {
   551→      startX: e.clientX,
   552→      startY: e.clientY,
   553→      // Start from the currently-displayed (clamped) position so consecutive
   554→      // drags don't accumulate out-of-bounds offsets.
   555→      panX: clampedPan.x,
   556→      panY: clampedPan.y,
   557→    }
   558→  }
   559→  const onPointerMove = (e: React.PointerEvent) => {
   560→    if (!dragRef.current) return
   561→    const dx = e.clientX - dragRef.current.startX
   562→    const dy = e.clientY - dragRef.current.startY
   563→    setPan(clampPan({ x: dragRef.current.panX + dx, y: dragRef.current.panY + dy }))
   564→  }
   565→  const onPointerUp = (e: React.PointerEvent) => {
   566→    dragRef.current = null
   567→    try {
   568→      e.currentTarget.releasePointerCapture(e.pointerId)
   569→    } catch {
   570→      /* ignore — pointer may already be released */
   571→    }
   572→  }
   573→
   574→  const onZoomChange = (v: number[]) => {
   575→    const newZoom = v[0] / 100
   576→    setZoom(newZoom)
   577→    // Re-clamp pan for the new zoom level so the image never reveals empty
   578→    // space outside its bounds.
   579→    const newEffScale = fillScale * newZoom
   580→    const newScaledW = imgSize ? imgSize.nw * newEffScale : 0
   581→    const newScaledH = imgSize ? imgSize.nh * newEffScale : 0
   582→    const newMaxPanX = Math.max(0, (newScaledW - CROP_CONTAINER_SIZE) / 2)
   583→    const newMaxPanY = Math.max(0, (newScaledH - CROP_CONTAINER_SIZE) / 2)
   584→    setPan((p) => ({
   585→      x: Math.max(-newMaxPanX, Math.min(newMaxPanX, p.x)),
   586→      y: Math.max(-newMaxPanY, Math.min(newMaxPanY, p.y)),
   587→    }))
   588→  }
   589→
   590→  // Draw the visible square region to a 256×256 canvas and return a JPEG Blob.
   591→  const cropToBlob = (): Promise<Blob | null> => {
   592→    return new Promise((resolve) => {
   593→      const img = imgRef.current
   594→      if (!img || !imgSize) {
   595→        resolve(null)
   596→        return
   597→      }
   598→      // Visible region in displayed-image coords (after transform):
   599→      //   visX = (scaledW - C) / 2 - pan.x
   600→      //   visY = (scaledH - C) / 2 - pan.y
   601→      // Convert to natural coords by dividing by effScale.
   602→      const sx = ((scaledW - CROP_CONTAINER_SIZE) / 2 - clampedPan.x) / effScale
   603→      const sy = ((scaledH - CROP_CONTAINER_SIZE) / 2 - clampedPan.y) / effScale
   604→      const sSize = CROP_CONTAINER_SIZE / effScale
   605→      const canvas = document.createElement('canvas')
   606→      canvas.width = CROP_OUTPUT_SIZE
   607→      canvas.height = CROP_OUTPUT_SIZE
   608→      const ctx = canvas.getContext('2d')
   609→      if (!ctx) {
   610→        resolve(null)
   611→        return
   612→      }
   613→      // White background in case the source has transparency.
   614→      ctx.fillStyle = '#ffffff'
   615→      ctx.fillRect(0, 0, CROP_OUTPUT_SIZE, CROP_OUTPUT_SIZE)
   616→      ctx.drawImage(img, sx, sy, sSize, sSize, 0, 0, CROP_OUTPUT_SIZE, CROP_OUTPUT_SIZE)
   617→      canvas.toBlob((b) => resolve(b), 'image/jpeg', 0.92)
   618→    })
   619→  }
   620→
   621→  const apply = async () => {
   622→    if (applying) return
   623→    setApplying(true)
   624→    try {
   625→      const blob = await cropToBlob()
   626→      if (!blob) throw new Error('Could not crop image')
   627→      const file = new File([blob], 'avatar.jpg', { type: 'image/jpeg' })
   628→      const { file: uploaded } = await uploadFile(file)
   629→      await api('/api/users/me', {
   630→        method: 'PUT',
   631→        body: JSON.stringify({ photoUrl: uploaded.url }),
   632→      })
   633→      await refresh()
   634→      toast.success('Photo updated')
   635→      onClose()
   636→    } catch (e: any) {
   637→      toast.error(e?.message || 'Failed to update photo')
   638→    } finally {
   639→      setApplying(false)
   640→    }
   641→  }
   642→
   643→  return (
   644→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   645→      <DialogContent className="max-w-md gap-0 p-0">
   646→        <DialogHeader className="border-b border-border px-4 py-3">
   647→          <DialogTitle className="flex items-center gap-2 text-base">
   648→            <Crop size={16} className="text-primary" />
   649→            Crop photo
   650→          </DialogTitle>
   651→        </DialogHeader>
   652→        <div className="p-4">
   653→          <div
   654→            className="relative mx-auto grid touch-none select-none place-items-center overflow-hidden rounded-xl bg-zinc-900"
   655→            style={{ width: CROP_CONTAINER_SIZE, height: CROP_CONTAINER_SIZE }}
   656→            onPointerDown={onPointerDown}
   657→            onPointerMove={onPointerMove}
   658→            onPointerUp={onPointerUp}
   659→            onPointerCancel={onPointerUp}
   660→          >
   661→            {src && (
   662→              <img
   663→                ref={imgRef}
   664→                src={src}
   665→                alt="Crop preview"
   666→                onLoad={onImgLoad}
   667→                onError={onImgError}
   668→                draggable={false}
   669→                className="pointer-events-none max-w-none select-none"
   670→                style={{
   671→                  width: imgSize ? imgSize.nw * fillScale : 'auto',
   672→                  height: imgSize ? imgSize.nh * fillScale : 'auto',
   673→                  transform: `translate(${clampedPan.x}px, ${clampedPan.y}px) scale(${zoom})`,
   674→                  transformOrigin: 'center center',
   675→                  opacity: loaded ? 1 : 0,
   676→                }}
   677→              />
   678→            )}
   679→            {/* Crop overlay: square white border + dark shadow outside the
   680→                crop region. Pointer events disabled so the underlying image
   681→                receives the drag. */}
   682→            <div className="pointer-events-none absolute inset-0">
   683→              <div
   684→                className="absolute"
   685→                style={{
   686→                  left: '50%',
   687→                  top: '50%',
   688→                  width: CROP_CONTAINER_SIZE,
   689→                  height: CROP_CONTAINER_SIZE,
   690→                  transform: 'translate(-50%, -50%)',
   691→                  boxShadow: '0 0 0 9999px rgba(0,0,0,0.5)',
   692→                  border: '2px solid white',
   693→                }}
   694→              />
   695→            </div>
   696→            {!loaded && (
   697→              <div className="absolute inset-0 grid place-items-center">
   698→                <Loader2 className="animate-spin text-white/80" size={24} />
   699→              </div>
   700→            )}
   701→          </div>
   702→
   703→          <div className="mt-4 space-y-3">
   704→            <div>
   705→              <div className="mb-1 flex items-center justify-between text-xs text-muted-foreground">
   706→                <span>Zoom</span>
   707→                <span>{zoom.toFixed(1)}×</span>
   708→              </div>
   709→              <Slider
   710→                value={[Math.round(zoom * 100)]}
   711→                min={100}
   712→                max={300}
   713→                step={1}
   714→                onValueChange={onZoomChange}
   715→                disabled={!loaded}
   716→              />
   717→            </div>
   718→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   719→              Drag to reposition. The crop is always square — what you see in
   720→              the box is what becomes your photo.
   721→            </p>
   722→          </div>
   723→        </div>
   724→        <DialogFooter className="flex-row justify-end gap-2 border-t border-border px-4 py-3">
   725→          <Button
   726→            type="button"
   727→            variant="ghost"
   728→            onClick={onClose}
   729→            disabled={applying}
   730→          >
   731→            <X size={16} className="mr-1" />
   732→            Cancel
   733→          </Button>
   734→          <Button
   735→            type="button"
   736→            onClick={apply}
   737→            disabled={!loaded || applying}
   738→          >
   739→            {applying ? (
   740→              <Loader2 size={16} className="mr-1 animate-spin" />
   741→            ) : (
   742→              <Check size={16} className="mr-1" />
   743→            )}
   744→            Apply
   745→          </Button>
   746→        </DialogFooter>
   747→      </DialogContent>
   748→    </Dialog>
   749→  )
   750→}
