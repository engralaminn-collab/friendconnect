   146→function formatBytes(bytes: number | null | undefined): string {
   147→  if (bytes === null || bytes === undefined) return ''
   148→  if (bytes === 0) return '0 B'
   149→  const units = ['B', 'KB', 'MB', 'GB']
   150→  let n = bytes
   151→  let i = 0
   152→  while (n >= 1024 && i < units.length - 1) {
   153→    n /= 1024
   154→    i++
   155→  }
   156→  return `${n.toFixed(n < 10 && i > 0 ? 1 : 0)} ${units[i]}`
   157→}

   159→function fileTypeFromMime(mime: string | null | undefined): MessageType {
   160→  if (!mime) return 'document'
   161→  if (mime.startsWith('image/')) return 'image'
   162→  if (mime.startsWith('video/')) return 'video'
   163→  if (mime.startsWith('audio/')) return 'audio'
   164→  return 'document'
   165→}

   167→function linkify(text: string): React.ReactNode[] {
   168→  const parts = text.split(/(https?:\/\/[^\s]+)/g)
   169→  return parts.map((part, i) => {
   170→    if (/^https?:\/\//.test(part)) {
   171→      return (
   172→        <a key={i} href={part} target="_blank" rel="noreferrer" className="underline break-all">
   173→          {part}
   174→        </a>
   175→      )
   176→    }
   177→    return <span key={i}>{part}</span>
   178→  })
   179→}

   181→function toISO(d: any): string {
   182→  if (!d) return new Date().toISOString()
   183→  if (typeof d === 'string') return d
   184→  if (d instanceof Date) return d.toISOString()
   185→  try {
   186→    return new Date(d).toISOString()
   187→  } catch {
   188→    return new Date().toISOString()
   189→  }
   190→}

   196→function normalizeMessage(raw: any): Message {
   197→  return {
   198→    id: raw.id,
   199→    senderId: raw.senderId,
   200→    senderName: raw.senderName ?? raw.sender?.name ?? null,
   201→    senderPhoto: raw.senderPhoto ?? raw.sender?.profile?.photoUrl ?? null,
   202→    receiverId: raw.receiverId ?? null,
   203→    groupId: raw.groupId ?? null,
   204→    content: raw.content ?? '',
   205→    type: (raw.type ?? 'text') as MessageType,
   206→    fileUrl: raw.fileUrl ?? null,
   207→    fileName: raw.fileName ?? null,
   208→    fileSize: raw.fileSize ?? null,
   209→    mimeType: raw.mimeType ?? null,
   210→    status: (raw.status ?? 'SENT') as MessageStatus,
   211→    createdAt: toISO(raw.createdAt),
   212→    reactions: Array.isArray(raw.reactions) ? raw.reactions.map((r: any) => ({ emoji: r.emoji, userId: r.userId, userName: r.userName })) : [],
   213→    deletedFor: raw.deletedFor ?? '',
   214→    forwardedFrom: raw.forwardedFrom ?? null,
   215→  }
   216→}

   218→// A message is rendered as the "🚫 This message was deleted" placeholder when
   219→// it was deleted for everyone OR explicitly deleted-for-me (deletedFor is a
   220→// comma-separated list of user ids, or the literal 'everyone').
   221→function isDeletedForMe(msg: Message, meId?: string): boolean {
   222→  if (!msg.deletedFor) return false
   223→  if (msg.deletedFor === 'everyone') return true
   224→  if (!meId) return false
   225→  return msg.deletedFor
   226→    .split(',')
   227→    .map((s) => s.trim())
   228→    .filter(Boolean)
   229→    .includes(meId)
   230→}

   232→function lastMessagePreview(c: Conversation, meId: string | undefined): string {
   233→  if (!c.lastMessage) return 'Tap to start chatting'
   234→  const lm = c.lastMessage
   235→  const prefix = meId && lm.senderId === meId ? 'You: ' : ''
   236→  if (lm.type === 'image') return `${prefix}📷 Photo`
   237→  if (lm.type === 'video') return `${prefix}🎥 Video`
   238→  if (lm.type === 'audio') return `${prefix}🎤 Audio`
   239→  if (lm.type === 'document') return `${prefix}📄 Document`
   240→  return `${prefix}${lm.content}`
   241→}

   260→// ---------------- conversation list ----------------
   261→function ConversationList() {
   262→  const [tab, setTab] = useState<'chats' | 'groups'>('chats')
   263→  const [query, setQuery] = useState('')
   264→  const [conversations, setConversations] = useState<Conversation[]>([])
   265→  const [groups, setGroups] = useState<GroupLite[]>([])
   266→  const [loading, setLoading] = useState(true)
   267→  const { openChat, openGroupChat } = useApp()
   268→  const { socket, isOnline } = useSocket()
   269→  const { user: me } = useAuth()
   270→  const meId = me?.id
   271→
   272→  const load = useCallback(async () => {
   273→    setLoading(true)
   274→    try {
   275→      const [c, g] = await Promise.all([
   276→        api<{ conversations: Conversation[] }>('/api/messages/conversations').catch(() => ({ conversations: [] as Conversation[] })),
   277→        api<{ groups: GroupLite[] }>('/api/groups').catch(() => ({ groups: [] as GroupLite[] })),
   278→      ])
   279→      setConversations(c.conversations)
   280→      setGroups(g.groups)
   281→    } catch (e: any) {
   282→      toast.error(e?.message || 'Failed to load')
   283→    } finally {
   284→      setLoading(false)
   285→    }
   286→  }, [])
   287→
   288→  useEffect(() => {
   289→    load()
   290→  }, [load])
   291→
   292→  // realtime: refresh lastMessage + unread when a new message arrives
   293→  useEffect(() => {
   294→    if (!socket || !meId) return
   295→    const onNew = (raw: any) => {
   296→      if (!raw || raw.groupId) return
   297→      const peerId = raw.senderId === meId ? raw.receiverId : raw.senderId
   298→      if (!peerId) return
   299→      setConversations((prev) => {
   300→        const exists = prev.find((c) => c.id === peerId)
   301→        const lm = {
   302→          content: raw.content || '',
   303→          type: raw.type || 'text',
   304→          senderId: raw.senderId,
   305→          createdAt: toISO(raw.createdAt),
   306→          status: raw.status || 'SENT',
   307→        }
   308→        const unreadInc = raw.receiverId === meId ? 1 : 0
   309→        if (exists) {
   310→          const updated: Conversation = {
   311→            ...exists,
   312→            lastMessage: lm,
   313→            unread: exists.unread + unreadInc,
   314→          }
   315→          return [updated, ...prev.filter((c) => c.id !== peerId)]
   316→        }
   317→        // unknown peer — refetch to populate name/photo
   318→        load()
   319→        return prev
   320→      })
   321→    }
   322→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   323→      setConversations((prev) =>
   324→        prev.map((c) =>
   325→          c.lastMessage && c.lastMessage.senderId === meId && c.lastMessage.status !== status
   326→            ? { ...c, lastMessage: { ...c.lastMessage, status } }
   327→            : c
   328→        )
   329→      )
   330→    }
   331→    const onGroupNew = (raw: any) => {
   332→      if (!raw?.groupId) return
   333→      // bump group to top by re-fetching once (lightweight)
   334→      load()
   335→    }
   336→    socket.on(RT.MESSAGE_NEW, onNew)
   337→    socket.on(RT.MESSAGE_STATUS, onStatus as any)
   338→    socket.on(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   339→    return () => {
   340→      socket.off(RT.MESSAGE_NEW, onNew)
   341→      socket.off(RT.MESSAGE_STATUS, onStatus as any)
   342→      socket.off(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   343→    }
   344→  }, [socket, meId, load])
   345→
   346→  const q = query.trim().toLowerCase()
   347→  const filteredConvs = q
   348→    ? conversations.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   349→    : conversations
   350→  const filteredGroups = q
   351→    ? groups.filter((g) => g.name.toLowerCase().includes(q) || (g.description || '').toLowerCase().includes(q))
   352→    : groups
   353→
   354→  const totalUnread = conversations.reduce((s, c) => s + c.unread, 0)
   355→
   356→  return (
   357→    <div className="flex h-full flex-col">
   358→      <div className="border-b border-border px-4 py-3">
   359→        <div className="mb-3 flex items-center justify-between">
   360→          <h1 className="text-xl font-bold">Chats</h1>
   361→          {totalUnread > 0 && (
   362→            <span className="rounded-full bg-primary/15 px-2 py-0.5 text-xs font-semibold text-primary">
   363→              {totalUnread} unread
   364→            </span>
   365→          )}
   366→        </div>
   367→        <div className="relative">
   368→          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   369→          <Input
   370→            value={query}
   371→            onChange={(e) => setQuery(e.target.value)}
   372→            placeholder="Search chats and groups"
   373→            className="rounded-xl pl-9"
   374→          />
   375→        </div>
   376→      </div>
   377→
   378→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'chats' | 'groups')} className="flex flex-1 flex-col overflow-hidden">
   379→        <TabsList className="mx-4 mt-3 grid grid-cols-2 rounded-xl">
   380→          <TabsTrigger value="chats" className="text-xs">
   381→            Chats
   382→            {totalUnread > 0 && (
   383→              <span className="ml-1 grid h-4 min-w-4 place-items-center rounded-full bg-primary px-1 text-[9px] font-bold text-primary-foreground">
   384→                {totalUnread > 9 ? '9+' : totalUnread}
   385→              </span>
   386→            )}
   387→          </TabsTrigger>
   388→          <TabsTrigger value="groups" className="text-xs">
   389→            Groups
   390→            {groups.length > 0 && (
   391→              <span className="ml-1 text-[10px] font-normal text-muted-foreground">{groups.length}</span>
   392→            )}
   393→          </TabsTrigger>
   394→        </TabsList>
   395→
   396→        <TabsContent value="chats" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   397→          {loading ? (
   398→            <div className="space-y-1 p-2">
   399→              {[0, 1, 2, 3, 4].map((i) => (
   400→                <ConvRowSkeleton key={i} />
   401→              ))}
   402→            </div>
   403→          ) : filteredConvs.length === 0 ? (
   404→            <EmptyState
   405→              icon={<MessageCircle size={36} />}
   406→              title={q ? 'No matching chats' : 'No conversations yet'}
   407→              description={q ? 'Try a different search.' : 'Add a friend and say hello to start chatting.'}
   408→            />
   409→          ) : (
   410→            <div className="space-y-0.5">
   411→              {filteredConvs.map((c) => (
   412→                <ConversationRow
   413→                  key={c.id}
   414→                  conversation={c}
   415→                  online={isOnline(c.id)}
   416→                  meId={meId}
   417→                  onOpen={() => openChat(c.id)}
   418→                  onTogglePin={async () => {
   419→                    try {
   420→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, pinned: !c.pinned }) })
   421→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, pinned: !x.pinned } : x)))
   422→                    } catch (e: any) { toast.error(e.message) }
   423→                  }}
   424→                  onToggleMute={async () => {
   425→                    try {
   426→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, muted: !c.muted }) })
   427→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, muted: !x.muted } : x)))
   428→                    } catch (e: any) { toast.error(e.message) }
   429→                  }}
   430→                />
   431→              ))}
   432→            </div>
   433→          )}
   434→        </TabsContent>
   435→
   436→        <TabsContent value="groups" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   437→          {loading ? (
   438→            <div className="space-y-1 p-2">
   439→              {[0, 1, 2, 3].map((i) => (
   440→                <ConvRowSkeleton key={i} />
   441→              ))}
   442→            </div>
   443→          ) : filteredGroups.length === 0 ? (
   444→            <EmptyState
   445→              icon={<Users size={36} />}
   446→              title={q ? 'No matching groups' : 'No groups yet'}
   447→              description={q ? 'Try a different search.' : 'Create a group from the Groups tab to chat with multiple friends.'}
   448→            />
   449→          ) : (
   450→            <div className="space-y-0.5">
   451→              {filteredGroups.map((g) => (
   452→                <GroupRow key={g.id} group={g} onOpen={() => openGroupChat(g.id)} />
   453→              ))}
   454→            </div>
   455→          )}
   456→        </TabsContent>
   457→      </Tabs>
   458→    </div>
   459→  )
   460→}

   462→function ConversationRow({
   463→  conversation,
   464→  online,
   465→  meId,
   466→  onOpen,
   467→  onTogglePin,
   468→  onToggleMute,
   469→}: {
   470→  conversation: Conversation
   471→  online: boolean
   472→  meId?: string
   473→  onOpen: () => void
   474→  onTogglePin: () => void
   475→  onToggleMute: () => void
   476→}) {
   477→  const [menuOpen, setMenuOpen] = useState(false)
   478→  const pressTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   479→  const [pinned, setPinned] = useState(!!conversation.pinned)
   480→  const [muted, setMuted] = useState(!!conversation.muted)
   481→
   482→  useEffect(() => {
   483→    setPinned(!!conversation.pinned)
   484→    setMuted(!!conversation.muted)
   485→  }, [conversation.pinned, conversation.muted])
   486→
   487→  const startPress = () => {
   488→    pressTimer.current = setTimeout(() => setMenuOpen(true), 500)
   489→  }
   490→  const cancelPress = () => {
   491→    if (pressTimer.current) clearTimeout(pressTimer.current)
   492→  }
   493→
   494→  return (
   495→    <div className="relative">
   496→      <button
   497→        onClick={onOpen}
   498→        onPointerDown={startPress}
   499→        onPointerUp={cancelPress}
   500→        onPointerLeave={cancelPress}
   501→        className={cn(
   502→          'flex w-full items-center gap-3 rounded-xl p-2.5 text-left transition-colors hover:bg-accent/50',
   503→          pinned && 'bg-primary/5'
   504→        )}
   505→      >
   506→        <UserAvatar name={conversation.name} src={conversation.photoUrl} size={48} online={online} />
   507→        <div className="min-w-0 flex-1">
   508→          <div className="flex items-baseline justify-between gap-2">
   509→            <span className="flex min-w-0 items-center gap-1 truncate text-sm font-semibold">
   510→              <span className="truncate">{conversation.name}</span>
   511→              {pinned && <Pin size={11} className="shrink-0 text-primary" />}
   512→              {muted && <BellOff size={11} className="shrink-0 text-muted-foreground" />}
   513→            </span>
   514→            {conversation.lastMessage && (
   515→              <span className="shrink-0 text-[10px] text-muted-foreground">
   516→                {timeAgo(conversation.lastMessage.createdAt)}
   517→              </span>
   518→            )}
   519→          </div>
   520→          <div className="flex items-center justify-between gap-2">
   521→            <span
   522→              className={cn(
   523→                'truncate text-xs',
   524→                conversation.unread > 0 && !muted ? 'font-medium text-foreground' : 'text-muted-foreground'
   525→              )}
   526→            >
   527→              {lastMessagePreview(conversation, meId)}
   528→            </span>
   529→            {conversation.unread > 0 && (
   530→              <span
   531→                className={cn(
   532→                  'grid h-5 min-w-5 shrink-0 place-items-center rounded-full px-1.5 text-[10px] font-bold',
   533→                  muted ? 'bg-muted text-muted-foreground' : 'bg-primary text-primary-foreground'
   534→                )}
   535→              >
   536→                {conversation.unread > 9 ? '9+' : conversation.unread}
   537→              </span>
   538→            )}
   539→          </div>
   540→        </div>
   541→      </button>
   542→      {menuOpen && (
   543→        <>
   544→          <div className="fixed inset-0 z-20" onClick={() => setMenuOpen(false)} />
   545→          <div className="absolute right-2 top-12 z-30 w-44 overflow-hidden rounded-xl border border-border bg-popover p-1 shadow-xl fc-rise">
   546→            <button
   547→              onClick={() => { onTogglePin(); setPinned((p) => !p); setMenuOpen(false) }}
   548→              className="flex w-full items-center gap-2 rounded-lg px-3 py-2 text-sm hover:bg-accent"
   549→            >

   606→// ---------------- chat room (direct + group) ----------------
   607→function ChatRoom({
   608→  mode,
   609→  peerId,
   610→  onBack,
   611→}: {
   612→  mode: 'direct' | 'group'
   613→  peerId: string
   614→  onBack: () => void
   615→}) {
   616→  const { user: me } = useAuth()
   617→  const { socket, isOnline } = useSocket()
   618→  const [messages, setMessages] = useState<Message[]>([])
   619→  const [peerInfo, setPeerInfo] = useState<PeerInfo | null>(null)
   620→  const [loading, setLoading] = useState(true)
   621→  const [text, setText] = useState('')
   622→  const [peerTyping, setPeerTyping] = useState(false)
   623→  const [uploading, setUploading] = useState(false)
   624→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   625→  const [loadingOlder, setLoadingOlder] = useState(false)
   626→  const [hasMore, setHasMore] = useState(true)
   627→  const [searchQuery, setSearchQuery] = useState('')
   628→  const [searchOpen, setSearchOpen] = useState(false)
   629→  const [recording, setRecording] = useState(false)
   630→  const [recordSecs, setRecordSecs] = useState(0)
   631→  const [deleteTarget, setDeleteTarget] = useState<Message | null>(null)
   632→  const [forwardTarget, setForwardTarget] = useState<Message | null>(null)
   633→
   634→  const scrollRef = useRef<HTMLDivElement>(null)
   635→  const bottomRef = useRef<HTMLDivElement>(null)
   636→  const fileInputRef = useRef<HTMLInputElement>(null)
   637→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   638→  const atBottomRef = useRef(true)
   639→  const oldestTsRef = useRef<string | null>(null)
   640→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   641→  const recordedChunksRef = useRef<Blob[]>([])
   642→  const recordStreamRef = useRef<MediaStream | null>(null)
   643→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   644→  const prevHeightRef = useRef(0)
   645→  const skipAutoScrollRef = useRef(false)
   646→  const meId = me?.id
   647→
   648→  // ---- load history + peer info ----
   649→  useEffect(() => {
   650→    let cancelled = false
   651→    setLoading(true)
   652→    setMessages([])
   653→    setHasMore(true)
   654→    setPeerTyping(false)
   655→    setPeerInfo(null)
   656→    oldestTsRef.current = null
   657→    atBottomRef.current = true
   658→
   659→    const endpoint = `/api/messages?${mode === 'direct' ? 'userId' : 'groupId'}=${encodeURIComponent(peerId)}`
   660→
   661→    Promise.all([
   662→      api<{ messages: Message[] }>(endpoint)
   663→        .then((d) => d.messages)
   664→        .catch((e: any) => {
   665→          toast.error(e?.message || 'Failed to load messages')
   666→          return [] as Message[]
   667→        }),
   668→      mode === 'direct'
   669→        ? api<{ friends: Friend[] }>('/api/friends')
   670→            .then((d) => d.friends.find((f) => f.id === peerId) || null)
   671→            .then((friend): PeerInfo | null => {
   672→              if (friend) {
   673→                return {
   674→                  name: friend.name,
   675→                  photoUrl: friend.photoUrl,
   676→                  username: friend.username,
   677→                  lastSeen: friend.lastSeen,
   678→                }
   679→              }
   680→              // fallback to conversations endpoint (peer may not be a friend yet but has messages)
   681→              return api<{ conversations: Conversation[] }>('/api/messages/conversations')
   682→                .then((d) => {
   683→                  const c = d.conversations.find((c) => c.id === peerId)
   684→                  return c
   685→                    ? { name: c.name, photoUrl: c.photoUrl, username: c.username, lastSeen: null }
   686→                    : null
   687→                })
   688→                .catch(() => null)
   689→            })
   690→            .catch(() => null)
   691→        : api<{ groups: GroupLite[] }>('/api/groups')
   692→            .then((d) => {
   693→              const g = d.groups.find((g) => g.id === peerId)
   694→              return g
   695→                ? { name: g.name, photoUrl: g.photoUrl, memberCount: g.memberCount }
   696→                : null
   697→            })
   698→            .catch(() => null),
   699→    ])
   700→      .then(([msgs, info]) => {
   701→        if (cancelled) return
   702→        const normalized = msgs.map(normalizeMessage)
   703→        setMessages(normalized)
   704→        if (normalized.length > 0) oldestTsRef.current = normalized[0].createdAt
   705→        if (normalized.length < 50) setHasMore(false)
   706→        if (info) setPeerInfo(info)
   707→      })
   708→      .finally(() => {
   709→        if (!cancelled) setLoading(false)
   710→      })
   711→
   712→    return () => {
   713→      cancelled = true
   714→    }
   715→  }, [mode, peerId])
   716→
   717→  // ---- jump to bottom after first load ----
   718→  useEffect(() => {
   719→    if (loading) return
   720→    const id = requestAnimationFrame(() => {
   721→      bottomRef.current?.scrollIntoView({ block: 'end' })
   722→    })
   723→    return () => cancelAnimationFrame(id)
   724→  }, [loading])
   725→
   726→  // ---- auto-scroll on new messages ----
   727→  useEffect(() => {
   728→    if (loading) return
   729→    if (skipAutoScrollRef.current) {
   730→      skipAutoScrollRef.current = false
   731→      return
   732→    }
   733→    if (loadingOlder) return
   734→    if (!atBottomRef.current) return
   735→    const id = requestAnimationFrame(() => {
   736→      bottomRef.current?.scrollIntoView({ block: 'end', behavior: 'smooth' })
   737→    })
   738→    return () => cancelAnimationFrame(id)
   739→  }, [messages.length, peerTyping, loading, loadingOlder])
   740→
   741→  // ---- socket subscriptions ----
   742→  useEffect(() => {
   743→    if (!socket || !meId) return
   744→
   745→    const isRelevant = (raw: any): boolean => {
   746→      if (!raw) return false
   747→      if (mode === 'direct') {
   748→        if (raw.groupId) return false
   749→        return (
   750→          (raw.senderId === peerId && raw.receiverId === meId) ||
   751→          (raw.senderId === meId && raw.receiverId === peerId)
   752→        )
   753→      }
   754→      return raw.groupId === peerId
   755→    }
   756→
   757→    const onNew = (raw: any) => {
   758→      if (!isRelevant(raw)) return
   759→      const msg = normalizeMessage(raw)
   760→      setMessages((prev) => {
   761→        if (prev.some((m) => m.id === msg.id)) return prev
   762→        return [...prev, msg]
   763→      })
   764→      // mark realtime incoming 1:1 messages as seen
   765→      if (mode === 'direct' && msg.senderId === peerId && msg.receiverId === meId) {
   766→        socket.emit(RT.MESSAGE_SEEN, { messageId: msg.id })
   767→      }
   768→    }
   769→
   770→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   771→      setMessages((prev) =>
   772→        prev.map((m) => (m.id === id ? { ...m, status: status as MessageStatus } : m))
   773→      )
   774→    }
   775→
   776→    const onTyping = ({ userId, isTyping }: { userId: string; isTyping: boolean }) => {
   777→      if (mode === 'direct' && userId === peerId) setPeerTyping(isTyping)
   778→    }
   779→
   780→    const onReaction = ({ messageId, userId, emoji, action }: { messageId: string; userId: string; emoji: string; action: 'add' | 'remove' }) => {
   781→      setMessages((prev) =>
   782→        prev.map((m) => {
   783→          if (m.id !== messageId) return m
   784→          const reactions = m.reactions ? [...m.reactions] : []
   785→          if (action === 'add') {
   786→            if (!reactions.some((r) => r.userId === userId && r.emoji === emoji)) {
   787→              reactions.push({ emoji, userId, userName: '' })
   788→            }
   789→          } else {
   790→            const idx = reactions.findIndex((r) => r.userId === userId && r.emoji === emoji)
   791→            if (idx >= 0) reactions.splice(idx, 1)
   792→          }
   793→          return { ...m, reactions }
   794→        })
   795→      )
   796→    }
   797→
   798→    // Realtime message delete: when another participant deletes a message for
   799→    // everyone, the server emits message:deleted { messageId, scope:'everyone' }.
   800→    // We swap the local message for the "🚫 This message was deleted" placeholder.
   801→    const onDeleted = ({ messageId, scope }: { messageId: string; scope: string }) => {
   802→      if (scope !== 'everyone') return
   803→      setMessages((prev) =>
   804→        prev.map((m) =>
   805→          m.id === messageId
   806→            ? {
   807→                ...m,
   808→                deletedFor: 'everyone',
   809→                content: '',
   810→                fileUrl: null,
   811→                fileName: null,
   812→                fileSize: null,
   813→                mimeType: null,
   814→                type: 'text',
   815→                reactions: [],
   816→              }
   817→            : m
   818→        )
   819→      )
   820→    }
   821→
   822→    socket.on(RT.MESSAGE_NEW, onNew)
   823→    socket.on(RT.GROUP_MESSAGE_NEW, onNew)
   824→    socket.on(RT.MESSAGE_STATUS, onStatus)
   825→    socket.on(RT.TYPING_INDICATOR, onTyping)
   826→    socket.on(RT.REACTION_UPDATE, onReaction)
   827→    socket.on(RT.MESSAGE_DELETED, onDeleted)
   828→
   829→    return () => {
   830→      socket.off(RT.MESSAGE_NEW, onNew)
   831→      socket.off(RT.GROUP_MESSAGE_NEW, onNew)
   832→      socket.off(RT.MESSAGE_STATUS, onStatus)
   833→      socket.off(RT.TYPING_INDICATOR, onTyping)
   834→      socket.off(RT.REACTION_UPDATE, onReaction)
   835→      socket.off(RT.MESSAGE_DELETED, onDeleted)
   836→      if (mode === 'direct') {
   837→        socket.emit(RT.TYPING, { receiverId: peerId, isTyping: false })
   838→      }
   839→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current)
   840→    }
   841→  }, [socket, meId, peerId, mode])
   842→
   843→  // ---- toggle reaction on a message ----
   844→  const toggleReaction = useCallback(
   845→    (messageId: string, emoji: string) => {
   846→      if (!socket) return
   847→      setMessages((prev) =>
   848→        prev.map((m) => {
   849→          if (m.id !== messageId) return m
   850→          const reactions = m.reactions ? [...m.reactions] : []
   851→          const existing = reactions.find((r) => r.userId === meId && r.emoji === emoji)
   852→          if (existing) {
   853→            const idx = reactions.indexOf(existing)
   854→            reactions.splice(idx, 1)
   855→            socket.emit(RT.REACTION_REMOVE, { messageId, emoji })
   856→          } else {
   857→            reactions.push({ emoji, userId: meId, userName: me?.name || '' })
   858→            socket.emit(RT.REACTION_ADD, { messageId, emoji })
   859→          }
   860→          return { ...m, reactions }
   861→        })
   862→      )
   863→    },
   864→    [socket, meId, me]
   865→  )
   866→
   867→  // ---- reply (inserts a simple @reply prefix into the input) ----
   868→  const handleReply = useCallback((msg: Message) => {
   869→    setText((t) => (t.startsWith('@reply') ? t : `@reply ${t}`))
   870→  }, [])
   871→
   872→  // ---- delete message (for me / for everyone) ----
   873→  const confirmDelete = useCallback(
   874→    async (msg: Message, scope: 'me' | 'everyone') => {
   875→      setDeleteTarget(null)
   876→      try {
   877→        await api(`/api/messages/${msg.id}/delete`, {
   878→          method: 'POST',
   879→          body: JSON.stringify({ scope }),
   880→        })
   881→        if (scope === 'me') {
   882→          // optimistic: remove from local view immediately
   883→          setMessages((prev) => prev.filter((m) => m.id !== msg.id))
   884→          toast.success('Message deleted for you')
   885→        } else {
   886→          // tell other participants via realtime + swap local for placeholder
   887→          if (socket) {
   888→            socket.emit(RT.MESSAGE_DELETE, { messageId: msg.id, scope: 'everyone' })
   889→          }
   890→          setMessages((prev) =>
   891→            prev.map((m) =>
   892→              m.id === msg.id
   893→                ? {
   894→                    ...m,
   895→                    deletedFor: 'everyone',
   896→                    content: '',
   897→                    fileUrl: null,
   898→                    fileName: null,
   899→                    fileSize: null,
   900→                    mimeType: null,
   901→                    type: 'text',
   902→                    reactions: [],
   903→                  }
   904→                : m
   905→            )
   906→          )
   907→          toast.success('Message deleted for everyone')
   908→        }
   909→      } catch (e: any) {
   910→        toast.error(e?.message || 'Failed to delete message')
   911→      }
   912→    },
   913→    [socket]
   914→  )
   915→
   916→  // ---- scroll handling + infinite scroll up ----
   917→  const loadOlder = useCallback(async () => {
   918→    if (!oldestTsRef.current) return
   919→    setLoadingOlder(true)
   920→    const el = scrollRef.current
   921→    prevHeightRef.current = el?.scrollHeight || 0
   922→    try {
   923→      const endpoint = `/api/messages?${mode === 'direct' ? 'userId' : 'groupId'}=${encodeURIComponent(peerId)}&before=${encodeURIComponent(oldestTsRef.current)}`
   924→      const data = await api<{ messages: Message[] }>(endpoint)
   925→      if (data.messages.length === 0) {
   926→        setHasMore(false)
   927→      } else {
   928→        const normalized = data.messages.map(normalizeMessage)
   929→        skipAutoScrollRef.current = true
   930→        setMessages((prev) => [...normalized, ...prev])
   931→        oldestTsRef.current = normalized[0]?.createdAt || oldestTsRef.current
   932→        if (normalized.length < 50) setHasMore(false)
   933→        // restore scroll position after DOM update
   934→        requestAnimationFrame(() => {
   935→          if (el) {
   936→            const newHeight = el.scrollHeight
   937→            el.scrollTop = newHeight - prevHeightRef.current
   938→          }
   939→        })
   940→      }
   941→    } catch (e: any) {
   942→      toast.error(e?.message || 'Failed to load older messages')
   943→    } finally {
   944→      setLoadingOlder(false)
   945→    }
   946→  }, [mode, peerId])
   947→
   948→  const handleScroll = useCallback(() => {
   949→    const el = scrollRef.current
   950→    if (!el) return
   951→    const distFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
   952→    atBottomRef.current = distFromBottom < 80
   953→    if (el.scrollTop < 60 && hasMore && !loadingOlder && !loading && oldestTsRef.current) {
   954→      loadOlder()
   955→    }
   956→  }, [hasMore, loadingOlder, loading, loadOlder])
   957→
   958→  // ---- send text ----
   959→  const sendMessage = useCallback(
   960→    (partial: Pick<Message, 'content' | 'type' | 'fileUrl' | 'fileName' | 'fileSize' | 'mimeType'>) => {
   961→      if (!socket || !meId) return
   962→      const tempId = newTempId()
   963→      const optimistic: Message = {
   964→        id: tempId,
   965→        tempId,
   966→        senderId: meId,
   967→        senderName: me?.name || null,
   968→        senderPhoto: me?.profile?.photoUrl ?? null,
   969→        receiverId: mode === 'direct' ? peerId : null,
   970→        groupId: mode === 'group' ? peerId : null,
   971→        content: partial.content,
   972→        type: partial.type,
   973→        fileUrl: partial.fileUrl ?? null,
   974→        fileName: partial.fileName ?? null,
   975→        fileSize: partial.fileSize ?? null,
   976→        mimeType: partial.mimeType ?? null,
   977→        status: 'SENDING',
   978→        createdAt: new Date().toISOString(),
   979→      }
   980→      setMessages((prev) => [...prev, optimistic])
   981→      atBottomRef.current = true
   982→
   983→      const payload: any = {
   984→        tempId,
   985→        content: partial.content,
   986→        type: partial.type,
   987→        fileUrl: partial.fileUrl ?? undefined,
   988→        fileName: partial.fileName ?? undefined,
   989→        fileSize: partial.fileSize ?? undefined,
   990→        mimeType: partial.mimeType ?? undefined,
   991→        ...(mode === 'direct' ? { receiverId: peerId } : { groupId: peerId }),
   992→      }
   993→      const event = mode === 'direct' ? RT.MESSAGE_SEND : RT.GROUP_MESSAGE_SEND
   994→      socket.emit(event, payload, (ack: any) => {
   995→        if (ack?.ok && ack.message) {
   996→          const realMsg: Message = {
   997→            ...normalizeMessage(ack.message),
   998→            tempId,
   999→            senderName: ack.message.senderName ?? ack.message.sender?.name ?? me?.name ?? null,
  1000→            senderPhoto: ack.message.senderPhoto ?? ack.message.sender?.profile?.photoUrl ?? me?.profile?.photoUrl ?? null,
  1001→          }
  1002→          setMessages((prev) => prev.map((m) => (m.tempId === tempId ? realMsg : m)))
  1003→        } else {
  1004→          // leave optimistic in place with SENT fallback so user can retry
  1005→          setMessages((prev) =>
  1006→            prev.map((m) => (m.tempId === tempId ? { ...m, status: 'SENT' } : m))
  1007→          )
  1008→          toast.error(ack?.error || 'Failed to send message')
  1009→        }
  1010→      })
  1011→    },
  1012→    [socket, meId, me, mode, peerId]
  1013→  )
  1014→
  1015→  const sendText = useCallback(() => {
  1016→    const content = text.trim()
  1017→    if (!content) return
  1018→    sendMessage({ content, type: 'text', fileUrl: null, fileName: null, fileSize: null, mimeType: null })
  1019→    setText('')
  1020→    if (socket && mode === 'direct') {
  1021→      socket.emit(RT.TYPING, { receiverId: peerId, isTyping: false })
  1022→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current)
  1023→    }
  1024→  }, [text, sendMessage, socket, mode, peerId])
  1025→
  1026→  // ---- send file ----
  1027→  const sendFile = useCallback(
  1028→    async (file: File) => {
  1029→      setUploading(true)
  1030→      try {
  1031→        const { file: uploaded } = await uploadFile(file)
  1032→        const type = fileTypeFromMime(uploaded.mimeType)
  1033→        sendMessage({
  1034→          content: '',
  1035→          type,
  1036→          fileUrl: uploaded.url,
  1037→          fileName: uploaded.name,
  1038→          fileSize: uploaded.size,
  1039→          mimeType: uploaded.mimeType,
  1040→        })
  1041→      } catch (e: any) {
  1042→        toast.error(e?.message || 'Upload failed')
  1043→      } finally {
  1044→        setUploading(false)
  1045→        if (fileInputRef.current) fileInputRef.current.value = ''
  1046→      }
  1047→    },
  1048→    [sendMessage]
  1049→  )
  1050→
  1051→  // ---- voice note recording ----
  1052→  const startRecording = useCallback(async () => {
  1053→    try {
  1054→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
  1055→      recordStreamRef.current = stream
  1056→      recordedChunksRef.current = []
  1057→      const mr = new MediaRecorder(stream)
  1058→      mr.ondataavailable = (e) => {
  1059→        if (e.data.size > 0) recordedChunksRef.current.push(e.data)
  1060→      }
  1061→      mr.onstop = async () => {
  1062→        const blob = new Blob(recordedChunksRef.current, { type: mr.mimeType || 'audio/webm' })
  1063→        const ext = (mr.mimeType || 'audio/webm').includes('mp4') ? 'm4a' : 'webm'
  1064→        const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type })
  1065→        await sendFile(file)
  1066→      }
  1067→      mr.start()
  1068→      mediaRecorderRef.current = mr
  1069→      setRecording(true)
  1070→      setRecordSecs(0)
  1071→      recordTimerRef.current = setInterval(() => setRecordSecs((s) => s + 1), 1000)
  1072→    } catch (e: any) {
  1073→      toast.error(e?.message || 'Microphone access denied')
  1074→    }
  1075→  }, [sendFile])

  1486→function highlight(text: string, q: string): React.ReactNode[] {
  1487→  if (!q.trim()) return linkify(text)
  1488→  const lower = text.toLowerCase()
  1489→  const ql = q.toLowerCase()
  1490→  const nodes: React.ReactNode[] = []
  1491→  let i = 0
  1492→  let idx = lower.indexOf(ql, i)
  1493→  let key = 0
  1494→  while (idx !== -1) {
  1495→    if (idx > i) nodes.push(<span key={key++}>{text.slice(i, idx)}</span>)
  1496→    nodes.push(
  1497→      <mark key={key++} className="rounded bg-yellow-300 px-0.5 text-black">
  1498→        {text.slice(idx, idx + ql.length)}
  1499→      </mark>
  1500→    )
  1501→    i = idx + ql.length
  1502→    idx = lower.indexOf(ql, i)
  1503→  }
  1504→  if (i < text.length) nodes.push(<span key={key++}>{text.slice(i)}</span>)
  1505→  return nodes
  1506→}
