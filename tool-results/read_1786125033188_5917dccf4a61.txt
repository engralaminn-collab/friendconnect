   160→function formatBytes(bytes: number | null | undefined): string {
   161→  if (bytes === null || bytes === undefined) return ''
   162→  if (bytes === 0) return '0 B'
   163→  const units = ['B', 'KB', 'MB', 'GB']
   164→  let n = bytes
   165→  let i = 0
   166→  while (n >= 1024 && i < units.length - 1) {
   167→    n /= 1024
   168→    i++
   169→  }
   170→  return `${n.toFixed(n < 10 && i > 0 ? 1 : 0)} ${units[i]}`
   171→}

   173→function fileTypeFromMime(mime: string | null | undefined): MessageType {
   174→  if (!mime) return 'document'
   175→  if (mime.startsWith('image/')) return 'image'
   176→  if (mime.startsWith('video/')) return 'video'
   177→  if (mime.startsWith('audio/')) return 'audio'
   178→  return 'document'
   179→}

   181→function linkify(text: string): React.ReactNode[] {
   182→  const parts = text.split(/(https?:\/\/[^\s]+)/g)
   183→  return parts.map((part, i) => {
   184→    if (/^https?:\/\//.test(part)) {
   185→      return (
   186→        <a key={i} href={part} target="_blank" rel="noreferrer" className="underline break-all">
   187→          {part}
   188→        </a>
   189→      )
   190→    }
   191→    return <span key={i}>{part}</span>
   192→  })
   193→}

   195→function toISO(d: any): string {
   196→  if (!d) return new Date().toISOString()
   197→  if (typeof d === 'string') return d
   198→  if (d instanceof Date) return d.toISOString()
   199→  try {
   200→    return new Date(d).toISOString()
   201→  } catch {
   202→    return new Date().toISOString()
   203→  }
   204→}

   210→function normalizeMessage(raw: any): Message {
   211→  return {
   212→    id: raw.id,
   213→    senderId: raw.senderId,
   214→    senderName: raw.senderName ?? raw.sender?.name ?? null,
   215→    senderPhoto: raw.senderPhoto ?? raw.sender?.profile?.photoUrl ?? null,
   216→    receiverId: raw.receiverId ?? null,
   217→    groupId: raw.groupId ?? null,
   218→    content: raw.content ?? '',
   219→    type: (raw.type ?? 'text') as MessageType,
   220→    fileUrl: raw.fileUrl ?? null,
   221→    fileName: raw.fileName ?? null,
   222→    fileSize: raw.fileSize ?? null,
   223→    mimeType: raw.mimeType ?? null,
   224→    status: (raw.status ?? 'SENT') as MessageStatus,
   225→    createdAt: toISO(raw.createdAt),
   226→    reactions: Array.isArray(raw.reactions) ? raw.reactions.map((r: any) => ({ emoji: r.emoji, userId: r.userId, userName: r.userName })) : [],
   227→    deletedFor: raw.deletedFor ?? '',
   228→    forwardedFrom: raw.forwardedFrom ?? null,
   229→    starred: !!raw.starred,
   230→    expiresAt: raw.expiresAt ?? null,
   231→  }
   232→}

   234→// A message is rendered as the "🚫 This message was deleted" placeholder when
   235→// it was deleted for everyone OR explicitly deleted-for-me (deletedFor is a
   236→// comma-separated list of user ids, or the literal 'everyone').
   237→function isDeletedForMe(msg: Message, meId?: string): boolean {
   238→  if (!msg.deletedFor) return false
   239→  if (msg.deletedFor === 'everyone') return true
   240→  if (!meId) return false
   241→  return msg.deletedFor
   242→    .split(',')
   243→    .map((s) => s.trim())
   244→    .filter(Boolean)
   245→    .includes(meId)
   246→}

   248→function lastMessagePreview(c: Conversation, meId: string | undefined): string {
   249→  if (!c.lastMessage) return 'Tap to start chatting'
   250→  const lm = c.lastMessage
   251→  const prefix = meId && lm.senderId === meId ? 'You: ' : ''
   252→  if (lm.type === 'image') return `${prefix}📷 Photo`
   253→  if (lm.type === 'video') return `${prefix}🎥 Video`
   254→  if (lm.type === 'audio') return `${prefix}🎤 Audio`
   255→  if (lm.type === 'document') return `${prefix}📄 Document`
   256→  return `${prefix}${lm.content}`
   257→}

   276→// ---------------- conversation list ----------------
   277→function ConversationList() {
   278→  const [tab, setTab] = useState<'chats' | 'groups'>('chats')
   279→  const [query, setQuery] = useState('')
   280→  const [conversations, setConversations] = useState<Conversation[]>([])
   281→  const [archivedConvs, setArchivedConvs] = useState<Conversation[]>([])
   282→  const [drafts, setDrafts] = useState<Map<string, string>>(new Map())
   283→  const [archivedExpanded, setArchivedExpanded] = useState(false)
   284→  const [groups, setGroups] = useState<GroupLite[]>([])
   285→  const [loading, setLoading] = useState(true)
   286→  const { openChat, openGroupChat } = useApp()
   287→  const { socket, isOnline } = useSocket()
   288→  const { user: me } = useAuth()
   289→  const meId = me?.id
   290→
   291→  const load = useCallback(async () => {
   292→    setLoading(true)
   293→    try {
   294→      const [c, g, d] = await Promise.all([
   295→        api<{ conversations: Conversation[]; archived: Conversation[] }>(
   296→          '/api/messages/conversations'
   297→        ).catch(() => ({ conversations: [] as Conversation[], archived: [] as Conversation[] })),
   298→        api<{ groups: GroupLite[] }>('/api/groups').catch(() => ({ groups: [] as GroupLite[] })),
   299→        api<{ drafts: Array<{ peerId: string; isGroup: boolean; content: string; updatedAt: string }> }>(
   300→          '/api/messages/drafts'
   301→        ).catch(() => ({ drafts: [] as Array<{ peerId: string; isGroup: boolean; content: string; updatedAt: string }> })),
   302→      ])
   303→      setConversations(c.conversations)
   304→      setArchivedConvs(c.archived)
   305→      setGroups(g.groups)
   306→      const draftMap = new Map<string, string>()
   307→      for (const draft of d.drafts) {
   308→        if (draft.content) draftMap.set(draft.peerId, draft.content)
   309→      }
   310→      setDrafts(draftMap)
   311→    } catch (e: any) {
   312→      toast.error(e?.message || 'Failed to load')
   313→    } finally {
   314→      setLoading(false)
   315→    }
   316→  }, [])
   317→
   318→  useEffect(() => {
   319→    load()
   320→  }, [load])
   321→
   322→  // realtime: refresh lastMessage + unread when a new message arrives
   323→  useEffect(() => {
   324→    if (!socket || !meId) return
   325→    const onNew = (raw: any) => {
   326→      if (!raw || raw.groupId) return
   327→      const peerId = raw.senderId === meId ? raw.receiverId : raw.senderId
   328→      if (!peerId) return
   329→      setConversations((prev) => {
   330→        const exists = prev.find((c) => c.id === peerId)
   331→        const lm = {
   332→          content: raw.content || '',
   333→          type: raw.type || 'text',
   334→          senderId: raw.senderId,
   335→          createdAt: toISO(raw.createdAt),
   336→          status: raw.status || 'SENT',
   337→        }
   338→        const unreadInc = raw.receiverId === meId ? 1 : 0
   339→        if (exists) {
   340→          const updated: Conversation = {
   341→            ...exists,
   342→            lastMessage: lm,
   343→            unread: exists.unread + unreadInc,
   344→          }
   345→          return [updated, ...prev.filter((c) => c.id !== peerId)]
   346→        }
   347→        // unknown peer — refetch to populate name/photo
   348→        load()
   349→        return prev
   350→      })
   351→    }
   352→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   353→      setConversations((prev) =>
   354→        prev.map((c) =>
   355→          c.lastMessage && c.lastMessage.senderId === meId && c.lastMessage.status !== status
   356→            ? { ...c, lastMessage: { ...c.lastMessage, status } }
   357→            : c
   358→        )
   359→      )
   360→    }
   361→    const onGroupNew = (raw: any) => {
   362→      if (!raw?.groupId) return
   363→      // bump group to top by re-fetching once (lightweight)
   364→      load()
   365→    }
   366→    socket.on(RT.MESSAGE_NEW, onNew)
   367→    socket.on(RT.MESSAGE_STATUS, onStatus as any)
   368→    socket.on(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   369→    return () => {
   370→      socket.off(RT.MESSAGE_NEW, onNew)
   371→      socket.off(RT.MESSAGE_STATUS, onStatus as any)
   372→      socket.off(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   373→    }
   374→  }, [socket, meId, load])
   375→
   376→  const q = query.trim().toLowerCase()
   377→  const filteredConvs = q
   378→    ? conversations.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   379→    : conversations
   380→  const filteredArchived = q
   381→    ? archivedConvs.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   382→    : archivedConvs
   383→  const filteredGroups = q
   384→    ? groups.filter((g) => g.name.toLowerCase().includes(q) || (g.description || '').toLowerCase().includes(q))
   385→    : groups
   386→
   387→  const totalUnread = conversations.reduce((s, c) => s + c.unread, 0)
   388→
   389→  // ---- archive / unarchive a conversation (optimistic + API + toast) ----
   390→  const handleToggleArchive = useCallback(
   391→    async (c: Conversation, currentlyArchived: boolean) => {
   392→      const nextArchived = !currentlyArchived
   393→      // Optimistic move between active/archived arrays
   394→      if (nextArchived) {
   395→        setConversations((prev) => prev.filter((x) => x.id !== c.id))
   396→        setArchivedConvs((prev) => [{ ...c, archived: true }, ...prev])
   397→      } else {
   398→        setArchivedConvs((prev) => prev.filter((x) => x.id !== c.id))
   399→        setConversations((prev) => [...prev, { ...c, archived: false }])
   400→      }
   401→      try {
   402→        await api('/api/messages/preferences', {
   403→          method: 'POST',
   404→          body: JSON.stringify({ peerId: c.id, archived: nextArchived }),
   405→        })
   406→        toast.success(nextArchived ? 'Chat archived' : 'Chat unarchived')
   407→      } catch (e: any) {
   408→        toast.error(e?.message || 'Failed to update archive')
   409→        load()
   410→      }
   411→    },
   412→    [load]
   413→  )
   414→
   415→  return (
   416→    <div className="flex h-full flex-col">
   417→      <div className="border-b border-border px-4 py-3">
   418→        <div className="mb-3 flex items-center justify-between">
   419→          <h1 className="text-xl font-bold">Chats</h1>
   420→          {totalUnread > 0 && (
   421→            <span className="rounded-full bg-primary/15 px-2 py-0.5 text-xs font-semibold text-primary">
   422→              {totalUnread} unread
   423→            </span>
   424→          )}
   425→        </div>
   426→        <div className="relative">
   427→          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   428→          <Input
   429→            value={query}
   430→            onChange={(e) => setQuery(e.target.value)}
   431→            placeholder="Search chats and groups"
   432→            className="rounded-xl pl-9"
   433→          />
   434→        </div>
   435→      </div>
   436→
   437→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'chats' | 'groups')} className="flex flex-1 flex-col overflow-hidden">
   438→        <TabsList className="mx-4 mt-3 grid grid-cols-2 rounded-xl">
   439→          <TabsTrigger value="chats" className="text-xs">
   440→            Chats
   441→            {totalUnread > 0 && (
   442→              <span className="ml-1 grid h-4 min-w-4 place-items-center rounded-full bg-primary px-1 text-[9px] font-bold text-primary-foreground">
   443→                {totalUnread > 9 ? '9+' : totalUnread}
   444→              </span>
   445→            )}
   446→          </TabsTrigger>
   447→          <TabsTrigger value="groups" className="text-xs">
   448→            Groups
   449→            {groups.length > 0 && (
   450→              <span className="ml-1 text-[10px] font-normal text-muted-foreground">{groups.length}</span>
   451→            )}
   452→          </TabsTrigger>
   453→        </TabsList>
   454→
   455→        <TabsContent value="chats" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   456→          {loading ? (
   457→            <div className="space-y-1 p-2">
   458→              {[0, 1, 2, 3, 4].map((i) => (
   459→                <ConvRowSkeleton key={i} />
   460→              ))}
   461→            </div>
   462→          ) : filteredConvs.length === 0 && filteredArchived.length === 0 ? (
   463→            <EmptyState
   464→              icon={<MessageCircle size={36} />}
   465→              title={q ? 'No matching chats' : 'No conversations yet'}
   466→              description={q ? 'Try a different search.' : 'Add a friend and say hello to start chatting.'}
   467→            />
   468→          ) : (
   469→            <div className="space-y-0.5">
   470→              {filteredConvs.map((c) => (
   471→                <ConversationRow
   472→                  key={c.id}
   473→                  conversation={c}
   474→                  online={isOnline(c.id)}
   475→                  meId={meId}
   476→                  draftPreview={drafts.get(c.id)}
   477→                  onOpen={() => openChat(c.id)}
   478→                  onTogglePin={async () => {
   479→                    try {
   480→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, pinned: !c.pinned }) })
   481→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, pinned: !x.pinned } : x)))
   482→                    } catch (e: any) { toast.error(e.message) }
   483→                  }}
   484→                  onToggleMute={async () => {
   485→                    try {
   486→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, muted: !c.muted }) })
   487→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, muted: !x.muted } : x)))
   488→                    } catch (e: any) { toast.error(e.message) }
   489→                  }}
   490→                  onToggleArchive={() => handleToggleArchive(c, false)}
   491→                />
   492→              ))}
   493→
   494→              {/* Archived section (collapsed by default) */}
   495→              {filteredArchived.length > 0 && (
   496→                <div className="mt-4">
   497→                  <button
   498→                    type="button"
   499→                    onClick={() => setArchivedExpanded((v) => !v)}
   500→                    className="flex w-full items-center gap-1.5 px-2 py-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground transition hover:text-foreground"
   501→                    aria-expanded={archivedExpanded}
   502→                    aria-controls="archived-chats-list"
   503→                  >
   504→                    {archivedExpanded ? <ChevronDown size={12} /> : <ChevronRight size={12} />}
   505→                    Archived ({filteredArchived.length})
   506→                  </button>
   507→                  {archivedExpanded && (
   508→                    <div id="archived-chats-list" className="space-y-0.5">
   509→                      {filteredArchived.map((c) => (
   510→                        <ConversationRow
   511→                          key={c.id}
   512→                          conversation={c}
   513→                          online={isOnline(c.id)}
   514→                          meId={meId}
   515→                          draftPreview={drafts.get(c.id)}

   517→                          onOpen={() => openChat(c.id)}
   518→                          onTogglePin={async () => {

   748→function ChatRoom({
   749→  mode,
   750→  peerId,
   751→  onBack,
   752→}: {
   753→  mode: 'direct' | 'group'
   754→  peerId: string
   755→  onBack: () => void
   756→}) {
   757→  const { user: me } = useAuth()
   758→  const { socket, isOnline } = useSocket()
   759→  const [messages, setMessages] = useState<Message[]>([])
   760→  const [peerInfo, setPeerInfo] = useState<PeerInfo | null>(null)
   761→  const [loading, setLoading] = useState(true)
   762→  const [text, setText] = useState('')
   763→  const [peerTyping, setPeerTyping] = useState(false)
   764→  const [uploading, setUploading] = useState(false)
   765→  const [imagePreview, setImagePreview] = useState<string | null>(null)
   766→  const [loadingOlder, setLoadingOlder] = useState(false)
   767→  const [hasMore, setHasMore] = useState(true)
   768→  const [searchQuery, setSearchQuery] = useState('')
   769→  const [searchOpen, setSearchOpen] = useState(false)
   770→  const [recording, setRecording] = useState(false)
   771→  const [recordSecs, setRecordSecs] = useState(0)
   772→  const [deleteTarget, setDeleteTarget] = useState<Message | null>(null)
   773→  const [forwardTarget, setForwardTarget] = useState<Message | null>(null)
   774→  const [settingsOpen, setSettingsOpen] = useState(false)
   775→  const [starredDialogOpen, setStarredDialogOpen] = useState(false)
   776→  const [disappearing, setDisappearing] = useState<'OFF' | '24H' | '7D'>('OFF')
   777→  const [wallpaper, setWallpaper] = useState<string | null>(null)
   778→
   779→  const scrollRef = useRef<HTMLDivElement>(null)
   780→  const bottomRef = useRef<HTMLDivElement>(null)
   781→  const fileInputRef = useRef<HTMLInputElement>(null)
   782→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   783→  const atBottomRef = useRef(true)
   784→  const oldestTsRef = useRef<string | null>(null)
   785→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   786→  const recordedChunksRef = useRef<Blob[]>([])
   787→  const recordStreamRef = useRef<MediaStream | null>(null)
   788→  const recordTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   789→  const prevHeightRef = useRef(0)
   790→  const skipAutoScrollRef = useRef(false)
   791→  // Draft auto-save: a ref mirror of `text` so the peerId-change cleanup can
   792→  // read the *current* text without a stale closure, and a debounce timer.
   793→  const textRef = useRef('')
   794→  const draftSaveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   795→  const meId = me?.id
   796→
   797→  // keep textRef in sync with `text` (so cleanup reads the latest typed value)
   798→  useEffect(() => {
   799→    textRef.current = text
   800→  }, [text])
   801→
   802→  // ---- load history + peer info ----
   803→  useEffect(() => {
   804→    let cancelled = false
   805→    setLoading(true)
   806→    setMessages([])
   807→    setHasMore(true)
   808→    setPeerTyping(false)
   809→    setPeerInfo(null)
   810→    // Reset the input for the new conversation; the draft (if any) is restored
   811→    // below.
   812→    setText('')
   813→    // Cancel any pending draft autosave from the previous conversation.
   814→    if (draftSaveTimerRef.current) {
   815→      clearTimeout(draftSaveTimerRef.current)
   816→      draftSaveTimerRef.current = null
   817→    }
   818→    oldestTsRef.current = null
   819→    atBottomRef.current = true
   820→
   821→    const endpoint = `/api/messages?${mode === 'direct' ? 'userId' : 'groupId'}=${encodeURIComponent(peerId)}`
   822→
   823→    Promise.all([
   824→      api<{ messages: Message[] }>(endpoint)
   825→        .then((d) => d.messages)
   826→        .catch((e: any) => {
   827→          toast.error(e?.message || 'Failed to load messages')
   828→          return [] as Message[]
   829→        }),
   830→      mode === 'direct'
   831→        ? api<{ friends: Friend[] }>('/api/friends')
   832→            .then((d) => d.friends.find((f) => f.id === peerId) || null)
   833→            .then((friend): PeerInfo | null => {
   834→              if (friend) {
   835→                return {
   836→                  name: friend.name,
   837→                  photoUrl: friend.photoUrl,
   838→                  username: friend.username,
   839→                  lastSeen: friend.lastSeen,
   840→                }
   841→              }
   842→              // fallback to conversations endpoint (peer may not be a friend yet but has messages)
   843→              return api<{ conversations: Conversation[]; archived: Conversation[] }>('/api/messages/conversations')
   844→                .then((d) => {
   845→                  const c = d.conversations.find((c) => c.id === peerId) || d.archived.find((c) => c.id === peerId)
   846→                  return c
   847→                    ? { name: c.name, photoUrl: c.photoUrl, username: c.username, lastSeen: null }
   848→                    : null
   849→                })
   850→                .catch(() => null)
   851→            })
   852→            .catch(() => null)
   853→        : api<{ groups: GroupLite[] }>('/api/groups')
   854→            .then((d) => {
   855→              const g = d.groups.find((g) => g.id === peerId)
   856→              return g
   857→                ? { name: g.name, photoUrl: g.photoUrl, memberCount: g.memberCount }
   858→                : null
   859→            })
   860→            .catch(() => null),
   861→    ])
   862→      .then(([msgs, info]) => {
   863→        if (cancelled) return
   864→        const normalized = msgs.map(normalizeMessage)
   865→        setMessages(normalized)
   866→        if (normalized.length > 0) oldestTsRef.current = normalized[0].createdAt
   867→        if (normalized.length < 50) setHasMore(false)
   868→        if (info) setPeerInfo(info)
   869→      })
   870→      .finally(() => {
   871→        if (!cancelled) setLoading(false)
   872→      })
   873→
   874→    // Restore any saved draft for this peer (so the user picks up where they
   875→    // left off). isGroup is derived from `mode` for this same conversation.
   876→    const isGroup = mode === 'group'
   877→    api<{ draft: { content: string; updatedAt: string } | null }>(
   878→      `/api/messages/drafts?peerId=${encodeURIComponent(peerId)}&isGroup=${isGroup}`
   879→    )
   880→      .then((d) => {
   881→        if (cancelled) return
   882→        if (d.draft && d.draft.content) {
   883→          setText(d.draft.content)
   884→          toast.success('Draft restored', { duration: 1500 })
   885→        }
   886→      })
   887→      .catch(() => {})
   888→
   889→    return () => {
   890→      cancelled = true
   891→      // Save the draft one last time for the conversation we're leaving so it
   892→      // doesn't get lost. The closure captures the OLD peerId/mode; textRef
   893→      // holds the latest typed value at the moment of cleanup.
   894→      const leavingPeerId = peerId
   895→      const leavingIsGroup = isGroup
   896→      const content = textRef.current
   897→      if (draftSaveTimerRef.current) {
   898→        clearTimeout(draftSaveTimerRef.current)
   899→        draftSaveTimerRef.current = null
   900→      }
   901→      if (content.trim()) {
   902→        api('/api/messages/drafts', {
   903→          method: 'POST',
   904→          body: JSON.stringify({ peerId: leavingPeerId, isGroup: leavingIsGroup, content }),
   905→        }).catch(() => {})
   906→      }
   907→    }
   908→  }, [mode, peerId])
   909→
   910→  // ---- load chat preferences (disappearing + wallpaper) for this peer ----
   911→  useEffect(() => {
   912→    setDisappearing('OFF')
   913→    setWallpaper(null)
   914→    let cancelled = false
   915→    api<{ preferences: Array<{ peerId: string; disappearing: string; wallpaper: string | null }> }>(
   916→      '/api/messages/preferences'
   917→    )
   918→      .then((d) => {
   919→        if (cancelled) return
   920→        const p = d.preferences?.find((x) => x.peerId === peerId)
   921→        if (p) {
   922→          setDisappearing(p.disappearing === '24H' || p.disappearing === '7D' ? p.disappearing : 'OFF')
   923→          setWallpaper(p.wallpaper ?? null)
   924→        }
   925→      })
   926→      .catch(() => {})
   927→    return () => {
   928→      cancelled = true
   929→    }
   930→  }, [peerId])
   931→
   932→  // ---- jump to bottom after first load ----
   933→  useEffect(() => {
   934→    if (loading) return
   935→    const id = requestAnimationFrame(() => {
   936→      bottomRef.current?.scrollIntoView({ block: 'end' })
   937→    })
   938→    return () => cancelAnimationFrame(id)
   939→  }, [loading])
   940→
   941→  // ---- auto-scroll on new messages ----
   942→  useEffect(() => {
   943→    if (loading) return
   944→    if (skipAutoScrollRef.current) {
   945→      skipAutoScrollRef.current = false
   946→      return
   947→    }
   948→    if (loadingOlder) return
   949→    if (!atBottomRef.current) return
   950→    const id = requestAnimationFrame(() => {
   951→      bottomRef.current?.scrollIntoView({ block: 'end', behavior: 'smooth' })
   952→    })
   953→    return () => cancelAnimationFrame(id)
   954→  }, [messages.length, peerTyping, loading, loadingOlder])
   955→
   956→  // ---- socket subscriptions ----
   957→  useEffect(() => {
   958→    if (!socket || !meId) return
   959→
   960→    const isRelevant = (raw: any): boolean => {
   961→      if (!raw) return false
   962→      if (mode === 'direct') {
   963→        if (raw.groupId) return false
   964→        return (
   965→          (raw.senderId === peerId && raw.receiverId === meId) ||
   966→          (raw.senderId === meId && raw.receiverId === peerId)
   967→        )
   968→      }
   969→      return raw.groupId === peerId
   970→    }
   971→
   972→    const onNew = (raw: any) => {
   973→      if (!isRelevant(raw)) return
   974→      const msg = normalizeMessage(raw)
   975→      setMessages((prev) => {
   976→        if (prev.some((m) => m.id === msg.id)) return prev
   977→        return [...prev, msg]
   978→      })
   979→      // mark realtime incoming 1:1 messages as seen
   980→      if (mode === 'direct' && msg.senderId === peerId && msg.receiverId === meId) {
   981→        socket.emit(RT.MESSAGE_SEEN, { messageId: msg.id })
   982→      }
   983→    }
   984→
   985→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   986→      setMessages((prev) =>
   987→        prev.map((m) => (m.id === id ? { ...m, status: status as MessageStatus } : m))
   988→      )
   989→    }
   990→
   991→    const onTyping = ({ userId, isTyping }: { userId: string; isTyping: boolean }) => {
   992→      if (mode === 'direct' && userId === peerId) setPeerTyping(isTyping)
   993→    }
   994→
   995→    const onReaction = ({ messageId, userId, emoji, action }: { messageId: string; userId: string; emoji: string; action: 'add' | 'remove' }) => {
   996→      setMessages((prev) =>
   997→        prev.map((m) => {
   998→          if (m.id !== messageId) return m
   999→          const reactions = m.reactions ? [...m.reactions] : []
  1000→          if (action === 'add') {
  1001→            if (!reactions.some((r) => r.userId === userId && r.emoji === emoji)) {
  1002→              reactions.push({ emoji, userId, userName: '' })
  1003→            }
  1004→          } else {
  1005→            const idx = reactions.findIndex((r) => r.userId === userId && r.emoji === emoji)
  1006→            if (idx >= 0) reactions.splice(idx, 1)
  1007→          }
  1008→          return { ...m, reactions }
  1009→        })
  1010→      )
  1011→    }
  1012→
  1013→    // Realtime message delete: when another participant deletes a message for
  1014→    // everyone, the server emits message:deleted { messageId, scope:'everyone' }.
  1015→    // We swap the local message for the "🚫 This message was deleted" placeholder.
  1016→    const onDeleted = ({ messageId, scope }: { messageId: string; scope: string }) => {
  1017→      if (scope !== 'everyone') return
  1018→      setMessages((prev) =>
  1019→        prev.map((m) =>
  1020→          m.id === messageId
  1021→            ? {
  1022→                ...m,
  1023→                deletedFor: 'everyone',
  1024→                content: '',
  1025→                fileUrl: null,
  1026→                fileName: null,
  1027→                fileSize: null,
  1028→                mimeType: null,
  1029→                type: 'text',
  1030→                reactions: [],
  1031→              }
  1032→            : m
  1033→        )
  1034→      )
  1035→    }
  1036→
  1037→    socket.on(RT.MESSAGE_NEW, onNew)
  1038→    socket.on(RT.GROUP_MESSAGE_NEW, onNew)
  1039→    socket.on(RT.MESSAGE_STATUS, onStatus)
  1040→    socket.on(RT.TYPING_INDICATOR, onTyping)
  1041→    socket.on(RT.REACTION_UPDATE, onReaction)
  1042→    socket.on(RT.MESSAGE_DELETED, onDeleted)
  1043→
  1044→    return () => {
  1045→      socket.off(RT.MESSAGE_NEW, onNew)
  1046→      socket.off(RT.GROUP_MESSAGE_NEW, onNew)
  1047→      socket.off(RT.MESSAGE_STATUS, onStatus)
  1048→      socket.off(RT.TYPING_INDICATOR, onTyping)
  1049→      socket.off(RT.REACTION_UPDATE, onReaction)

  1828→function highlight(text: string, q: string): React.ReactNode[] {
