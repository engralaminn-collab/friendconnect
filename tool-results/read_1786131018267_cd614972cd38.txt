   480→   480→/**
   481→   481→ * Round 9: PhotoCropDialog — replaces the old "upload immediately" flow with
   482→   482→ * a client-side crop step. The parent picks an image and hands us its data
   483→   483→ * URL; we render it in a square viewport with a dark mask outside the crop
   484→   484→ * region, a 1×–3× zoom slider, and drag-to-pan. On Apply we draw the visible
   485→   485→ * square region to a 256×256 canvas, convert it to a JPEG Blob, upload it
   486→   486→ * via `uploadFile`, PUT the new URL to /api/users/me, and refresh the
   487→   487→ * session. The crop step is mandatory (no skip) so every avatar is square.
   488→   488→ */
   489→   489→const CROP_CONTAINER_SIZE = 280
   490→   490→const CROP_OUTPUT_SIZE = 256
   491→   491→
   492→   492→function PhotoCropDialog({
   493→   493→  src,
   494→   494→  onClose,
   495→   495→  refresh,
   496→   496→}: {
   497→   497→  src: string
   498→   498→  onClose: () => void
   499→   499→  refresh: () => Promise<AuthUser | null>
   500→   500→}) {
   501→   501→  const imgRef = useRef<HTMLImageElement | null>(null)
   502→   502→  const [loaded, setLoaded] = useState(false)
   503→   503→  const [imgSize, setImgSize] = useState<{ nw: number; nh: number } | null>(null)
   504→   504→  const [zoom, setZoom] = useState(1) // 1.0 – 3.0
   505→   505→  const [pan, setPan] = useState<{ x: number; y: number }>({ x: 0, y: 0 })
   506→   506→  const [applying, setApplying] = useState(false)
   507→   507→  // Pointer-drag state. Stored in a ref so move/up handlers can read the
   508→   508→  // latest values without re-creating the handlers each render.
   509→   509→  const dragRef = useRef<{ startX: number; startY: number; panX: number; panY: number } | null>(null)
   510→   510→
   511→   511→  // Reset transient state when a new image is handed in.
   512→   512→  useEffect(() => {
   513→   513→    setLoaded(false)
   514→   514→    setImgSize(null)
   515→   515→    setZoom(1)
   516→   516→    setPan({ x: 0, y: 0 })
   517→   517→  }, [src])
   518→   518→
   519→   519→  const onImgLoad = () => {
   520→   520→    const img = imgRef.current
   521→   521→    if (!img) return
   522→   522→    setImgSize({ nw: img.naturalWidth, nh: img.naturalHeight })
   523→   523→    setLoaded(true)
   524→   524→  }
   525→   525→  const onImgError = () => {
   526→   526→    toast.error('Could not load image')
   527→   527→    onClose()
   528→   528→  }
   529→   529→
   530→   530→  // fillScale = the scale at which the image just fills the container in its
   531→   531→  // smaller dimension (object-cover semantics). At zoom=1 the visible region
   532→   532→  // is the centered square of size min(nw,nh) in natural coords.
   533→   533→  const fillScale = imgSize
   534→   534→    ? CROP_CONTAINER_SIZE / Math.min(imgSize.nw, imgSize.nh)
   535→   535→    : 1
   536→   536→  const effScale = fillScale * zoom
   537→   537→  const scaledW = imgSize ? imgSize.nw * effScale : 0
   538→   538→  const scaledH = imgSize ? imgSize.nh * effScale : 0
   539→   539→  const maxPanX = Math.max(0, (scaledW - CROP_CONTAINER_SIZE) / 2)
   540→   540→  const maxPanY = Math.max(0, (scaledH - CROP_CONTAINER_SIZE) / 2)
   541→   541→  const clampPan = (p: { x: number; y: number }) => ({
   542→   542→    x: Math.max(-maxPanX, Math.min(maxPanX, p.x)),
   543→   543→    y: Math.max(-maxPanY, Math.min(maxPanY, p.y)),
   544→   544→  })
   545→   545→  const clampedPan = clampPan(pan)
   546→   546→
   547→   547→  const onPointerDown = (e: React.PointerEvent) => {
   548→   548→    if (!loaded) return
   549→   549→    e.currentTarget.setPointerCapture(e.pointerId)
   550→   550→    dragRef.current = {
   551→   551→      startX: e.clientX,
   552→   552→      startY: e.clientY,
   553→   553→      // Start from the currently-displayed (clamped) position so consecutive
   554→   554→      // drags don't accumulate out-of-bounds offsets.
   555→   555→      panX: clampedPan.x,
   556→   556→      panY: clampedPan.y,
   557→   557→    }
   558→   558→  }
   559→   559→  const onPointerMove = (e: React.PointerEvent) => {
   560→   560→    if (!dragRef.current) return
   561→   561→    const dx = e.clientX - dragRef.current.startX
   562→   562→    const dy = e.clientY - dragRef.current.startY
   563→   563→    setPan(clampPan({ x: dragRef.current.panX + dx, y: dragRef.current.panY + dy }))
   564→   564→  }
   565→   565→  const onPointerUp = (e: React.PointerEvent) => {
   566→   566→    dragRef.current = null
   567→   567→    try {
   568→   568→      e.currentTarget.releasePointerCapture(e.pointerId)
   569→   569→    } catch {
   570→   570→      /* ignore — pointer may already be released */
   571→   571→    }
   572→   572→  }
   573→   573→
   574→   574→  const onZoomChange = (v: number[]) => {
   575→   575→    const newZoom = v[0] / 100
   576→   576→    setZoom(newZoom)
   577→   577→    // Re-clamp pan for the new zoom level so the image never reveals empty
   578→   578→    // space outside its bounds.
   579→   579→    const newEffScale = fillScale * newZoom
   580→   580→    const newScaledW = imgSize ? imgSize.nw * newEffScale : 0
   581→   581→    const newScaledH = imgSize ? imgSize.nh * newEffScale : 0
   582→   582→    const newMaxPanX = Math.max(0, (newScaledW - CROP_CONTAINER_SIZE) / 2)
   583→   583→    const newMaxPanY = Math.max(0, (newScaledH - CROP_CONTAINER_SIZE) / 2)
   584→   584→    setPan((p) => ({
   585→   585→      x: Math.max(-newMaxPanX, Math.min(newMaxPanX, p.x)),
   586→   586→      y: Math.max(-newMaxPanY, Math.min(newMaxPanY, p.y)),
   587→   587→    }))
   588→   588→  }
   589→   589→
   590→   590→  // Draw the visible square region to a 256×256 canvas and return a JPEG Blob.
   591→   591→  const cropToBlob = (): Promise<Blob | null> => {
   592→   592→    return new Promise((resolve) => {
   593→   593→      const img = imgRef.current
   594→   594→      if (!img || !imgSize) {
   595→   595→        resolve(null)
   596→   596→        return
   597→   597→      }
   598→   598→      // Visible region in displayed-image coords (after transform):
   599→   599→      //   visX = (scaledW - C) / 2 - pan.x
   600→   600→      //   visY = (scaledH - C) / 2 - pan.y
   601→   601→      // Convert to natural coords by dividing by effScale.
   602→   602→      const sx = ((scaledW - CROP_CONTAINER_SIZE) / 2 - clampedPan.x) / effScale
   603→   603→      const sy = ((scaledH - CROP_CONTAINER_SIZE) / 2 - clampedPan.y) / effScale
   604→   604→      const sSize = CROP_CONTAINER_SIZE / effScale
   605→   605→      const canvas = document.createElement('canvas')
   606→   606→      canvas.width = CROP_OUTPUT_SIZE
   607→   607→      canvas.height = CROP_OUTPUT_SIZE
   608→   608→      const ctx = canvas.getContext('2d')
   609→   609→      if (!ctx) {
   610→   610→        resolve(null)
   611→   611→        return
   612→   612→      }
   613→   613→      // White background in case the source has transparency.
   614→   614→      ctx.fillStyle = '#ffffff'
   615→   615→      ctx.fillRect(0, 0, CROP_OUTPUT_SIZE, CROP_OUTPUT_SIZE)
   616→   616→      ctx.drawImage(img, sx, sy, sSize, sSize, 0, 0, CROP_OUTPUT_SIZE, CROP_OUTPUT_SIZE)
   617→   617→      canvas.toBlob((b) => resolve(b), 'image/jpeg', 0.92)
   618→   618→    })
   619→   619→  }
   620→   620→
   621→   621→  const apply = async () => {
   622→   622→    if (applying) return
   623→   623→    setApplying(true)
   624→   624→    try {
   625→   625→      const blob = await cropToBlob()
   626→   626→      if (!blob) throw new Error('Could not crop image')
   627→   627→      const file = new File([blob], 'avatar.jpg', { type: 'image/jpeg' })
   628→   628→      const { file: uploaded } = await uploadFile(file)
   629→   629→      await api('/api/users/me', {
   630→   630→        method: 'PUT',
   631→   631→        body: JSON.stringify({ photoUrl: uploaded.url }),
   632→   632→      })
   633→   633→      await refresh()
   634→   634→      toast.success('Photo updated')
   635→   635→      onClose()
   636→   636→    } catch (e: any) {
   637→   637→      toast.error(e?.message || 'Failed to update photo')
   638→   638→    } finally {
   639→   639→      setApplying(false)
   640→   640→    }
   641→   641→  }
   642→   642→
   643→   643→  return (
   644→   644→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   645→   645→      <DialogContent className="max-w-md gap-0 p-0">
   646→   646→        <DialogHeader className="border-b border-border px-4 py-3">
   647→   647→          <DialogTitle className="flex items-center gap-2 text-base">
   648→   648→            <Crop size={16} className="text-primary" />
   649→   649→            Crop photo
   650→   650→          </DialogTitle>
   651→   651→        </DialogHeader>
   652→   652→        <div className="p-4">
   653→   653→          <div
   654→   654→            className="relative mx-auto grid touch-none select-none place-items-center overflow-hidden rounded-xl bg-zinc-900"
   655→   655→            style={{ width: CROP_CONTAINER_SIZE, height: CROP_CONTAINER_SIZE }}
   656→   656→            onPointerDown={onPointerDown}
   657→   657→            onPointerMove={onPointerMove}
   658→   658→            onPointerUp={onPointerUp}
   659→   659→            onPointerCancel={onPointerUp}
   660→   660→          >
   661→   661→            {src && (
   662→   662→              <img
   663→   663→                ref={imgRef}
   664→   664→                src={src}
   665→   665→                alt="Crop preview"
   666→   666→                onLoad={onImgLoad}
   667→   667→                onError={onImgError}
   668→   668→                draggable={false}
   669→   669→                className="pointer-events-none max-w-none select-none"
   670→   670→                style={{
   671→   671→                  width: imgSize ? imgSize.nw * fillScale : 'auto',
   672→   672→                  height: imgSize ? imgSize.nh * fillScale : 'auto',
   673→   673→                  transform: `translate(${clampedPan.x}px, ${clampedPan.y}px) scale(${zoom})`,
   674→   674→                  transformOrigin: 'center center',
   675→   675→                  opacity: loaded ? 1 : 0,
   676→   676→                }}
   677→   677→              />
   678→   678→            )}
   679→   679→            {/* Crop overlay: square white border + dark shadow outside the
   680→   680→                crop region. Pointer events disabled so the underlying image
   681→   681→                receives the drag. */}
   682→   682→            <div className="pointer-events-none absolute inset-0">
   683→   683→              <div
   684→   684→                className="absolute"
   685→   685→                style={{
   686→   686→                  left: '50%',
   687→   687→                  top: '50%',
   688→   688→                  width: CROP_CONTAINER_SIZE,
   689→   689→                  height: CROP_CONTAINER_SIZE,
   690→   690→                  transform: 'translate(-50%, -50%)',
   691→   691→                  boxShadow: '0 0 0 9999px rgba(0,0,0,0.5)',
   692→   692→                  border: '2px solid white',
   693→   693→                }}
   694→   694→              />
   695→   695→            </div>
   696→   696→            {!loaded && (
   697→   697→              <div className="absolute inset-0 grid place-items-center">
   698→   698→                <Loader2 className="animate-spin text-white/80" size={24} />
   699→   699→              </div>
   700→   700→            )}
   701→   701→          </div>
   702→   702→
   703→   703→          <div className="mt-4 space-y-3">
   704→   704→            <div>
   705→   705→              <div className="mb-1 flex items-center justify-between text-xs text-muted-foreground">
   706→   706→                <span>Zoom</span>
   707→   707→                <span>{zoom.toFixed(1)}×</span>
   708→   708→              </div>
   709→   709→              <Slider
   710→   710→                value={[Math.round(zoom * 100)]}
   711→   711→                min={100}
   712→   712→                max={300}
   713→   713→                step={1}
   714→   714→                onValueChange={onZoomChange}
   715→   715→                disabled={!loaded}
   716→   716→              />
   717→   717→            </div>
   718→   718→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   719→   719→              Drag to reposition. The crop is always square — what you see in
   720→   720→              the box is what becomes your photo.
   721→   721→            </p>
   722→   722→          </div>
   723→   723→        </div>
   724→   724→        <DialogFooter className="flex-row justify-end gap-2 border-t border-border px-4 py-3">
   725→   725→          <Button
   726→   726→            type="button"
   727→   727→            variant="ghost"
   728→   728→            onClick={onClose}
   729→   729→            disabled={applying}
   730→   730→          >
   731→   731→            <X size={16} className="mr-1" />
   732→   732→            Cancel
   733→   733→          </Button>
   734→   734→          <Button
   735→   735→            type="button"
   736→   736→            onClick={apply}
   737→   737→            disabled={!loaded || applying}
   738→   738→          >
   739→   739→            {applying ? (
   740→   740→              <Loader2 size={16} className="mr-1 animate-spin" />
   741→   741→            ) : (
   742→   742→              <Check size={16} className="mr-1" />
   743→   743→            )}
   744→   744→            Apply
   745→   745→          </Button>
   746→   746→        </DialogFooter>
   747→   747→      </DialogContent>
   748→   748→    </Dialog>
   749→   749→  )
   750→   750→}
