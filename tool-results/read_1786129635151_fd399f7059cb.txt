   197→function formatBytes(bytes: number | null | undefined): string {
   198→  if (bytes === null || bytes === undefined) return ''
   199→  if (bytes === 0) return '0 B'
   200→  const units = ['B', 'KB', 'MB', 'GB']
   201→  let n = bytes
   202→  let i = 0
   203→  while (n >= 1024 && i < units.length - 1) {
   204→    n /= 1024
   205→    i++
   206→  }
   207→  return `${n.toFixed(n < 10 && i > 0 ? 1 : 0)} ${units[i]}`
   208→}

   210→function fileTypeFromMime(mime: string | null | undefined): MessageType {
   211→  if (!mime) return 'document'
   212→  if (mime.startsWith('image/')) return 'image'
   213→  if (mime.startsWith('video/')) return 'video'
   214→  if (mime.startsWith('audio/')) return 'audio'
   215→  return 'document'
   216→}

   220→function isPdfMessage(msg: Message): boolean {
   221→  if (msg.mimeType && msg.mimeType.toLowerCase() === 'application/pdf') return true

   223→  return false

   226→function linkify(text: string): React.ReactNode[] {
   227→  const parts = text.split(/(https?:\/\/[^\s]+)/g)
   228→  return parts.map((part, i) => {
   229→    if (/^https?:\/\//.test(part)) {
   230→      return (
   231→        <a key={i} href={part} target="_blank" rel="noreferrer" className="underline break-all">
   232→          {part}
   233→        </a>
   234→      )
   235→    }
   236→    return <span key={i}>{part}</span>
   237→  })
   238→}

   240→function toISO(d: any): string {
   241→  if (!d) return new Date().toISOString()
   242→  if (typeof d === 'string') return d
   243→  if (d instanceof Date) return d.toISOString()
   244→  try {
   245→    return new Date(d).toISOString()
   246→  } catch {
   247→    return new Date().toISOString()
   248→  }
   249→}

   255→// Round 8: format an ISO date for the scheduled-messages viewer/list,
   256→// e.g. "Jan 5, 3:42 PM". Used by the toast + the ScheduledMessagesDialog rows.
   257→function formatScheduledTime(iso: string | null | undefined): string {
   258→  if (!iso) return ''
   259→  const d = new Date(iso)
   260→  if (isNaN(d.getTime())) return ''
   261→  return d.toLocaleString([], {
   262→    month: 'short',
   263→    day: 'numeric',
   264→    hour: '2-digit',
   265→    minute: '2-digit',
   266→  })
   267→}

   269→// Round 8: convert a Date to the YYYY-MM-DDTHH:MM format required by
   270→// <input type="datetime-local">. The value is in the user's local timezone
   271→// (the browser interprets datetime-local values as local time, not UTC).
   272→function toLocalDateTimeInput(d: Date): string {
   273→  const pad = (n: number) => String(n).padStart(2, '0')
   274→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   275→}

   305→// A message is rendered as the "🚫 This message was deleted" placeholder when
   306→// it was deleted for everyone OR explicitly deleted-for-me (deletedFor is a
   307→// comma-separated list of user ids, or the literal 'everyone').
   308→function isDeletedForMe(msg: Message, meId?: string): boolean {
   309→  if (!msg.deletedFor) return false
   310→  if (msg.deletedFor === 'everyone') return true
   311→  if (!meId) return false
   312→  return msg.deletedFor
   313→    .split(',')
   314→    .map((s) => s.trim())
   315→    .filter(Boolean)
   316→    .includes(meId)
   317→}

   319→function lastMessagePreview(c: Conversation, meId: string | undefined): string {
   320→  if (!c.lastMessage) return 'Tap to start chatting'
   321→  const lm = c.lastMessage
   322→  const prefix = meId && lm.senderId === meId ? 'You: ' : ''
   323→  if (lm.type === 'image') return `${prefix}📷 Photo`
   324→  if (lm.type === 'video') return `${prefix}🎥 Video`
   325→  if (lm.type === 'audio') return `${prefix}🎤 Audio`
   326→  if (lm.type === 'document') return `${prefix}📄 Document`
   327→  return `${prefix}${lm.content}`
   328→}

   347→// ---------------- conversation list ----------------
   348→function ConversationList() {
   349→  const [tab, setTab] = useState<'chats' | 'groups'>('chats')
   350→  const [query, setQuery] = useState('')
   351→  const [conversations, setConversations] = useState<Conversation[]>([])
   352→  const [archivedConvs, setArchivedConvs] = useState<Conversation[]>([])
   353→  const [drafts, setDrafts] = useState<Map<string, string>>(new Map())
   354→  const [archivedExpanded, setArchivedExpanded] = useState(false)
   355→  const [groups, setGroups] = useState<GroupLite[]>([])
   356→  const [loading, setLoading] = useState(true)
   357→  // Round 6: global message search state — fires debounced (400ms) when the
   358→  // search box has text. Results render in the search dropdown alongside the
   359→  // local conversation filter.
   360→  const [globalResults, setGlobalResults] = useState<GlobalSearchResult[]>([])
   361→  const [globalLoading, setGlobalLoading] = useState(false)
   362→  const globalSearchTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   363→  // Round 7: typing-status in conversation list. Tracks which conversations
   364→  // currently have someone typing (peerId for 1:1, `group:${groupId}` for
   365→  // groups). Each entry has a 3s auto-expiry timer so a missed isTyping:false
   366→  // doesn't leave a stale "typing…" indicator forever.
   367→  const [typingConvs, setTypingConvs] = useState<Set<string>>(new Set())
   368→  const typingTimersRef = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map())
   369→  const { openChat, openGroupChat } = useApp()
   370→  const { socket, isOnline } = useSocket()
   371→  const { user: me } = useAuth()
   372→  const meId = me?.id
   373→
   374→  const load = useCallback(async () => {
   375→    setLoading(true)
   376→    try {
   377→      const [c, g, d] = await Promise.all([
   378→        api<{ conversations: Conversation[]; archived: Conversation[] }>(
   379→          '/api/messages/conversations'
   380→        ).catch(() => ({ conversations: [] as Conversation[], archived: [] as Conversation[] })),
   381→        api<{ groups: GroupLite[] }>('/api/groups').catch(() => ({ groups: [] as GroupLite[] })),
   382→        api<{ drafts: Array<{ peerId: string; isGroup: boolean; content: string; updatedAt: string }> }>(
   383→          '/api/messages/drafts'
   384→        ).catch(() => ({ drafts: [] as Array<{ peerId: string; isGroup: boolean; content: string; updatedAt: string }> })),
   385→      ])
   386→      setConversations(c.conversations)
   387→      setArchivedConvs(c.archived)
   388→      setGroups(g.groups)
   389→      const draftMap = new Map<string, string>()
   390→      for (const draft of d.drafts) {
   391→        if (draft.content) draftMap.set(draft.peerId, draft.content)
   392→      }
   393→      setDrafts(draftMap)
   394→    } catch (e: any) {
   395→      toast.error(e?.message || 'Failed to load')
   396→    } finally {
   397→      setLoading(false)
   398→    }
   399→  }, [])
   400→
   401→  useEffect(() => {
   402→    load()
   403→  }, [load])
   404→
   405→  // realtime: refresh lastMessage + unread when a new message arrives
   406→  useEffect(() => {
   407→    if (!socket || !meId) return
   408→    const onNew = (raw: any) => {
   409→      if (!raw || raw.groupId) return
   410→      const peerId = raw.senderId === meId ? raw.receiverId : raw.senderId
   411→      if (!peerId) return
   412→      setConversations((prev) => {
   413→        const exists = prev.find((c) => c.id === peerId)
   414→        const lm = {
   415→          content: raw.content || '',
   416→          type: raw.type || 'text',
   417→          senderId: raw.senderId,
   418→          createdAt: toISO(raw.createdAt),
   419→          status: raw.status || 'SENT',
   420→        }
   421→        const unreadInc = raw.receiverId === meId ? 1 : 0
   422→        if (exists) {
   423→          const updated: Conversation = {
   424→            ...exists,
   425→            lastMessage: lm,
   426→            unread: exists.unread + unreadInc,
   427→          }
   428→          return [updated, ...prev.filter((c) => c.id !== peerId)]
   429→        }
   430→        // unknown peer — refetch to populate name/photo
   431→        load()
   432→        return prev
   433→      })
   434→    }
   435→    const onStatus = ({ id, status }: { id: string; status: string }) => {
   436→      setConversations((prev) =>
   437→        prev.map((c) =>
   438→          c.lastMessage && c.lastMessage.senderId === meId && c.lastMessage.status !== status
   439→            ? { ...c, lastMessage: { ...c.lastMessage, status } }
   440→            : c
   441→        )
   442→      )
   443→    }
   444→    const onGroupNew = (raw: any) => {
   445→      if (!raw?.groupId) return
   446→      // bump group to top by re-fetching once (lightweight)
   447→      load()
   448→    }
   449→    socket.on(RT.MESSAGE_NEW, onNew)
   450→    socket.on(RT.MESSAGE_STATUS, onStatus as any)
   451→    socket.on(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   452→    return () => {
   453→      socket.off(RT.MESSAGE_NEW, onNew)
   454→      socket.off(RT.MESSAGE_STATUS, onStatus as any)
   455→      socket.off(RT.GROUP_MESSAGE_NEW, onGroupNew as any)
   456→    }
   457→  }, [socket, meId, load])
   458→
   459→  // Round 7: typing-status in conversation list. Subscribe to both
   460→  // typing:indicator (1:1 — keyed by userId) and group:typing:indicator
   461→  // (keyed by `group:${groupId}`). Each "isTyping:true" event schedules a
   462→  // 3s auto-expiry timer so a missed isTyping:false (e.g. the peer's socket
   463→  // disconnected mid-typing) doesn't leave a stale "typing…" forever.
   464→  // Filters out the user's own id (the server already excludes the typer
   465→  // for group events; for 1:1 the typing:indicator is emitted to the
   466→  // receiver only, but the check is harmless + defensive).
   467→  useEffect(() => {
   468→    if (!socket || !meId) return
   469→
   470→    const clearTyping = (key: string) => {
   471→      setTypingConvs((prev) => {
   472→        if (!prev.has(key)) return prev
   473→        const next = new Set(prev)
   474→        next.delete(key)
   475→        return next
   476→      })
   477→    }
   478→
   479→    const scheduleExpiry = (key: string) => {
   480→      const timers = typingTimersRef.current
   481→      const existing = timers.get(key)
   482→      if (existing) clearTimeout(existing)
   483→      const t = setTimeout(() => {
   484→        timers.delete(key)
   485→        clearTyping(key)
   486→      }, 3000)
   487→      timers.set(key, t)
   488→    }
   489→
   490→    const onTyping = ({ userId, isTyping }: { userId: string; isTyping: boolean }) => {
   491→      if (!userId || userId === meId) return
   492→      if (isTyping) {
   493→        setTypingConvs((prev) => {
   494→          if (prev.has(userId)) return prev
   495→          const next = new Set(prev)
   496→          next.add(userId)
   497→          return next
   498→        })
   499→        scheduleExpiry(userId)
   500→      } else {
   501→        const t = typingTimersRef.current.get(userId)
   502→        if (t) {
   503→          clearTimeout(t)
   504→          typingTimersRef.current.delete(userId)
   505→        }
   506→        clearTyping(userId)
   507→      }
   508→    }
   509→
   510→    const onGroupTyping = ({
   511→      groupId,
   512→      userId,
   513→      isTyping,
   514→    }: {
   515→      groupId: string
   516→      userId: string
   517→      isTyping: boolean
   518→    }) => {
   519→      if (!groupId || userId === meId) return
   520→      const key = `group:${groupId}`
   521→      if (isTyping) {
   522→        setTypingConvs((prev) => {
   523→          if (prev.has(key)) return prev
   524→          const next = new Set(prev)
   525→          next.add(key)
   526→          return next
   527→        })
   528→        scheduleExpiry(key)
   529→      } else {
   530→        const t = typingTimersRef.current.get(key)
   531→        if (t) {
   532→          clearTimeout(t)
   533→          typingTimersRef.current.delete(key)
   534→        }
   535→        clearTyping(key)
   536→      }
   537→    }
   538→
   539→    socket.on(RT.TYPING_INDICATOR, onTyping)
   540→    socket.on(RT.GROUP_TYPING_INDICATOR, onGroupTyping)
   541→    return () => {
   542→      socket.off(RT.TYPING_INDICATOR, onTyping)
   543→      socket.off(RT.GROUP_TYPING_INDICATOR, onGroupTyping)
   544→      typingTimersRef.current.forEach((t) => clearTimeout(t))
   545→      typingTimersRef.current.clear()
   546→    }
   547→  }, [socket, meId])
   548→
   549→  // Round 6: global message search — debounced 400ms after the user stops
   550→  // typing. Fires GET /api/messages/search?q=QUERY and renders the results in
   551→  // the search dropdown alongside the local conversation filter. The effect
   552→  // cleanup clears the timer on every query change so only the latest query
   553→  // triggers a request. Empty query → empty results (no API call).
   554→  useEffect(() => {
   555→    const qq = query.trim()
   556→    if (!qq) {
   557→      setGlobalResults([])
   558→      setGlobalLoading(false)
   559→      return
   560→    }
   561→    setGlobalLoading(true)
   562→    if (globalSearchTimerRef.current) {
   563→      clearTimeout(globalSearchTimerRef.current)
   564→      globalSearchTimerRef.current = null
   565→    }
   566→    globalSearchTimerRef.current = setTimeout(async () => {
   567→      try {
   568→        const d = await api<{ results: GlobalSearchResult[] }>(
   569→          `/api/messages/search?q=${encodeURIComponent(qq)}`
   570→        )
   571→        setGlobalResults(d.results || [])
   572→      } catch {
   573→        setGlobalResults([])
   574→      } finally {
   575→        setGlobalLoading(false)
   576→      }
   577→    }, 400)
   578→    return () => {
   579→      if (globalSearchTimerRef.current) {
   580→        clearTimeout(globalSearchTimerRef.current)
   581→        globalSearchTimerRef.current = null
   582→      }
   583→    }
   584→  }, [query])
   585→
   586→  const q = query.trim().toLowerCase()
   587→  const filteredConvs = q
   588→    ? conversations.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   589→    : conversations
   590→  const filteredArchived = q
   591→    ? archivedConvs.filter((c) => c.name.toLowerCase().includes(q) || c.username.toLowerCase().includes(q))
   592→    : archivedConvs
   593→  const filteredGroups = q
   594→    ? groups.filter((g) => g.name.toLowerCase().includes(q) || (g.description || '').toLowerCase().includes(q))
   595→    : groups
   596→
   597→  const totalUnread = conversations.reduce((s, c) => s + c.unread, 0)
   598→
   599→  // ---- archive / unarchive a conversation (optimistic + API + toast) ----
   600→  const handleToggleArchive = useCallback(
   601→    async (c: Conversation, currentlyArchived: boolean) => {
   602→      const nextArchived = !currentlyArchived
   603→      // Optimistic move between active/archived arrays
   604→      if (nextArchived) {
   605→        setConversations((prev) => prev.filter((x) => x.id !== c.id))
   606→        setArchivedConvs((prev) => [{ ...c, archived: true }, ...prev])
   607→      } else {
   608→        setArchivedConvs((prev) => prev.filter((x) => x.id !== c.id))
   609→        setConversations((prev) => [...prev, { ...c, archived: false }])
   610→      }
   611→      try {
   612→        await api('/api/messages/preferences', {
   613→          method: 'POST',
   614→          body: JSON.stringify({ peerId: c.id, archived: nextArchived }),
   615→        })
   616→        toast.success(nextArchived ? 'Chat archived' : 'Chat unarchived')
   617→      } catch (e: any) {
   618→        toast.error(e?.message || 'Failed to update archive')
   619→        load()
   620→      }
   621→    },
   622→    [load]
   623→  )
   624→
   625→  return (
   626→    <div className="flex h-full flex-col">
   627→      <div className="border-b border-border px-4 py-3">
   628→        <div className="mb-3 flex items-center justify-between">
   629→          <h1 className="text-xl font-bold">Chats</h1>
   630→          {totalUnread > 0 && (
   631→            <span className="rounded-full bg-primary/15 px-2 py-0.5 text-xs font-semibold text-primary">
   632→              {totalUnread} unread
   633→            </span>
   634→          )}
   635→        </div>
   636→        <div className="relative">
   637→          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   638→          <Input
   639→            value={query}
   640→            onChange={(e) => setQuery(e.target.value)}
   641→            placeholder="Search chats and messages"
   642→            className="rounded-xl pl-9 pr-9"
   643→          />
   644→          {query && (
   645→            <button
   646→              type="button"
   647→              onClick={() => setQuery('')}
   648→              className="absolute right-2 top-1/2 grid h-6 w-6 -translate-y-1/2 place-items-center rounded-full text-muted-foreground transition hover:bg-accent hover:text-foreground"
   649→              aria-label="Clear search"
   650→            >
   651→              <X size={14} />
   652→            </button>
   653→          )}
   654→        </div>
   655→      </div>
   656→
   657→      {/* Round 6: when the search box has text, replace the chat/group tabs
   658→          with a unified search dropdown that shows TWO sections — the local
   659→          conversation filter (chats + groups whose name matches) AND global
   660→          message search results (GET /api/messages/search?q=...). Each result
   661→          row is tappable and routes the user into the right conversation. */}
   662→      {query.trim() ? (
   663→        <div className="fc-scroll flex-1 overflow-y-auto p-2">
   664→          {/* Conversations section */}
   665→          {(filteredConvs.length > 0 || filteredGroups.length > 0) && (
   666→            <div className="mb-2">
   667→              <div className="px-2 py-1 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   668→                Conversations
   669→              </div>
   670→              {filteredConvs.slice(0, 5).map((c) => (
   671→                <button
   672→                  key={c.id}
   673→                  type="button"
   674→                  onClick={() => {
   675→                    openChat(c.id)
   676→                    setQuery('')
   677→                  }}
   678→                  className="flex w-full items-center gap-3 rounded-xl p-2.5 text-left transition hover:bg-accent/40"
   679→                >
   680→                  <UserAvatar name={c.name} src={c.photoUrl} size={40} online={isOnline(c.id)} />
   681→                  <div className="min-w-0 flex-1">
   682→                    <div className="truncate text-sm font-medium">{c.name}</div>
   683→                    <div className="truncate text-xs text-muted-foreground">@{c.username}</div>
   684→                  </div>
   685→                </button>
   686→              ))}
   687→              {filteredGroups.slice(0, 5).map((g) => (
   688→                <button
   689→                  key={g.id}
   690→                  type="button"
   691→                  onClick={() => {
   692→                    openGroupChat(g.id)
   693→                    setQuery('')
   694→                  }}
   695→                  className="flex w-full items-center gap-3 rounded-xl p-2.5 text-left transition hover:bg-accent/40"
   696→                >
   697→                  {g.photoUrl ? (
   698→                    <div className="h-10 w-10 shrink-0 overflow-hidden rounded-full ring-1 ring-border">
   699→                      <img src={g.photoUrl} alt={g.name} className="h-full w-full object-cover" />
   700→                    </div>
   701→                  ) : (
   702→                    <div className="grid h-10 w-10 shrink-0 place-items-center rounded-full bg-primary/15 text-primary ring-1 ring-border">
   703→                      <Users size={18} />
   704→                    </div>
   705→                  )}
   706→                  <div className="min-w-0 flex-1">
   707→                    <div className="truncate text-sm font-medium">{g.name}</div>
   708→                    <div className="truncate text-xs text-muted-foreground">
   709→                      {g.memberCount} {g.memberCount === 1 ? 'member' : 'members'}
   710→                    </div>
   711→                  </div>
   712→                </button>
   713→              ))}
   714→            </div>
   715→          )}
   716→
   717→          {/* Messages section (global search) */}
   718→          <div>
   719→            <div className="flex items-center gap-1.5 px-2 py-1 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   720→              <MessageSquare size={11} />
   721→              Messages
   722→            </div>
   723→            {globalLoading ? (
   724→              <div className="flex justify-center py-4">
   725→                <Loader2 className="animate-spin text-muted-foreground" size={16} />
   726→              </div>
   727→            ) : globalResults.length === 0 ? (
   728→              <div className="px-2 py-3 text-center text-xs text-muted-foreground">
   729→                No messages found
   730→              </div>
   731→            ) : (
   732→              globalResults.map((r) => (
   733→                <button
   734→                  key={r.messageId}
   735→                  type="button"
   736→                  onClick={() => {
   737→                    if (r.conversationType === 'direct' && r.peerId) {
   738→                      openChat(r.peerId)
   739→                    } else if (r.conversationType === 'group' && r.groupId) {
   740→                      openGroupChat(r.groupId)
   741→                    }
   742→                    setQuery('')
   743→                  }}
   744→                  className="flex w-full items-start gap-2.5 rounded-xl p-2.5 text-left transition hover:bg-accent/40"
   745→                >
   746→                  <MessageSquare size={14} className="mt-0.5 shrink-0 text-muted-foreground" />
   747→                  <div className="min-w-0 flex-1">
   748→                    <div className="line-clamp-2 text-sm">
   749→                      {highlight(r.content || '', query.trim())}
   750→                    </div>
   751→                    <div className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
   752→                      <span className="truncate">{r.senderName || 'Unknown'}</span>
   753→                      <span className="shrink-0">·</span>
   754→                      <span className="truncate">
   755→                        {r.conversationType === 'direct' ? r.peerName || 'Direct' : r.groupName || 'Group'}
   756→                      </span>
   757→                      <span className="shrink-0">·</span>
   758→                      <span className="shrink-0">{timeAgo(r.createdAt)}</span>
   759→                    </div>
   760→                  </div>
   761→                </button>
   762→              ))
   763→            )}
   764→          </div>
   765→
   766→          {/* Empty state when nothing matches at all */}
   767→          {!loading &&
   768→            filteredConvs.length === 0 &&
   769→            filteredGroups.length === 0 &&
   770→            !globalLoading &&
   771→            globalResults.length === 0 && (
   772→              <div className="py-8">
   773→                <EmptyState
   774→                  icon={<Search size={36} />}
   775→                  title="No results"
   776→                  description={`Nothing matches "${query.trim()}" in your chats or messages.`}
   777→                />
   778→              </div>
   779→            )}
   780→        </div>
   781→      ) : (
   782→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'chats' | 'groups')} className="flex flex-1 flex-col overflow-hidden">
   783→        <TabsList className="mx-4 mt-3 grid grid-cols-2 rounded-xl">
   784→          <TabsTrigger value="chats" className="text-xs">
   785→            Chats
   786→            {totalUnread > 0 && (
   787→              <span className="ml-1 grid h-4 min-w-4 place-items-center rounded-full bg-primary px-1 text-[9px] font-bold text-primary-foreground">
   788→                {totalUnread > 9 ? '9+' : totalUnread}
   789→              </span>
   790→            )}
   791→          </TabsTrigger>
   792→          <TabsTrigger value="groups" className="text-xs">
   793→            Groups
   794→            {groups.length > 0 && (
   795→              <span className="ml-1 text-[10px] font-normal text-muted-foreground">{groups.length}</span>
   796→            )}
   797→          </TabsTrigger>
   798→        </TabsList>
   799→
   800→        <TabsContent value="chats" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   801→          {loading ? (
   802→            <div className="space-y-1 p-2">
   803→              {[0, 1, 2, 3, 4].map((i) => (
   804→                <ConvRowSkeleton key={i} />
   805→              ))}
   806→            </div>
   807→          ) : filteredConvs.length === 0 && filteredArchived.length === 0 ? (
   808→            <EmptyState
   809→              icon={<MessageCircle size={36} />}
   810→              title={q ? 'No matching chats' : 'No conversations yet'}
   811→              description={q ? 'Try a different search.' : 'Add a friend and say hello to start chatting.'}
   812→            />
   813→          ) : (
   814→            <div className="space-y-0.5">
   815→              {filteredConvs.map((c) => (
   816→                <ConversationRow
   817→                  key={c.id}
   818→                  conversation={c}
   819→                  online={isOnline(c.id)}
   820→                  meId={meId}
   821→                  draftPreview={drafts.get(c.id)}
   822→                  isTyping={typingConvs.has(c.id)}
   823→                  onOpen={() => openChat(c.id)}
   824→                  onTogglePin={async () => {
   825→                    try {
   826→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, pinned: !c.pinned }) })
   827→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, pinned: !x.pinned } : x)))
   828→                    } catch (e: any) { toast.error(e.message) }
   829→                  }}
   830→                  onToggleMute={async () => {
   831→                    try {
   832→                      await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, muted: !c.muted }) })
   833→                      setConversations((prev) => prev.map((x) => (x.id === c.id ? { ...x, muted: !x.muted } : x)))
   834→                    } catch (e: any) { toast.error(e.message) }
   835→                  }}
   836→                  onToggleArchive={() => handleToggleArchive(c, false)}
   837→                />
   838→              ))}
   839→
   840→              {/* Archived section (collapsed by default) */}
   841→              {filteredArchived.length > 0 && (
   842→                <div className="mt-4">
   843→                  <button
   844→                    type="button"
   845→                    onClick={() => setArchivedExpanded((v) => !v)}
   846→                    className="flex w-full items-center gap-1.5 px-2 py-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground transition hover:text-foreground"
   847→                    aria-expanded={archivedExpanded}
   848→                    aria-controls="archived-chats-list"
   849→                  >
   850→                    {archivedExpanded ? <ChevronDown size={12} /> : <ChevronRight size={12} />}
   851→                    Archived ({filteredArchived.length})
   852→                  </button>
   853→                  {archivedExpanded && (
   854→                    <div id="archived-chats-list" className="space-y-0.5">
   855→                      {filteredArchived.map((c) => (
   856→                        <ConversationRow
   857→                          key={c.id}
   858→                          conversation={c}
   859→                          online={isOnline(c.id)}
   860→                          meId={meId}
   861→                          draftPreview={drafts.get(c.id)}
   862→                          isTyping={typingConvs.has(c.id)}
   863→                          archived
   864→                          onOpen={() => openChat(c.id)}
   865→                          onTogglePin={async () => {
   866→                            try {
   867→                              await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, pinned: !c.pinned }) })
   868→                              setArchivedConvs((prev) => prev.map((x) => (x.id === c.id ? { ...x, pinned: !x.pinned } : x)))
   869→                            } catch (e: any) { toast.error(e.message) }
   870→                          }}
   871→                          onToggleMute={async () => {
   872→                            try {
   873→                              await api('/api/messages/preferences', { method: 'POST', body: JSON.stringify({ peerId: c.id, muted: !c.muted }) })
   874→                              setArchivedConvs((prev) => prev.map((x) => (x.id === c.id ? { ...x, muted: !x.muted } : x)))
   875→                            } catch (e: any) { toast.error(e.message) }
   876→                          }}
   877→                          onToggleArchive={() => handleToggleArchive(c, true)}
   878→                        />
   879→                      ))}
   880→                    </div>
   881→                  )}
   882→                </div>
   883→              )}
   884→            </div>
   885→          )}
   886→        </TabsContent>
   887→
   888→        <TabsContent value="groups" className="fc-scroll m-0 flex-1 overflow-y-auto p-2">
   889→          {loading ? (
   890→            <div className="space-y-1 p-2">
   891→              {[0, 1, 2, 3].map((i) => (
   892→                <ConvRowSkeleton key={i} />
   893→              ))}
   894→            </div>
   895→          ) : filteredGroups.length === 0 ? (
   896→            <EmptyState
   897→              icon={<Users size={36} />}
   898→              title={q ? 'No matching groups' : 'No groups yet'}
   899→              description={q ? 'Try a different search.' : 'Create a group from the Groups tab to chat with multiple friends.'}
   900→            />
   901→          ) : (
   902→            <div className="space-y-0.5">
   903→              {filteredGroups.map((g) => (
   904→                <GroupRow
   905→                  key={g.id}
   906→                  group={g}
   907→                  isTyping={typingConvs.has(`group:${g.id}`)}
   908→                  onOpen={() => openGroupChat(g.id)}
   909→                />
   910→              ))}
   911→            </div>
   912→          )}
   913→        </TabsContent>
   914→      </Tabs>
   915→      )}
   916→    </div>
   917→  )
   918→}
   919→
   920→function ConversationRow({
   921→  conversation,
   922→  online,
   923→  meId,
   924→  draftPreview,
   925→  isTyping,
   926→  archived,
   927→  onOpen,
   928→  onTogglePin,
   929→  onToggleMute,
   930→  onToggleArchive,
   931→}: {
   932→  conversation: Conversation
   933→  online: boolean
   934→  meId?: string
   935→  draftPreview?: string
   936→  isTyping?: boolean
   937→  archived?: boolean
   938→  onOpen: () => void
   939→  onTogglePin: () => void
   940→  onToggleMute: () => void
   941→  onToggleArchive: () => void
   942→}) {
   943→  const [menuOpen, setMenuOpen] = useState(false)
   944→  const pressTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   945→  const [pinned, setPinned] = useState(!!conversation.pinned)
   946→  const [muted, setMuted] = useState(!!conversation.muted)
   947→
   948→  useEffect(() => {
   949→    setPinned(!!conversation.pinned)
   950→    setMuted(!!conversation.muted)
   951→  }, [conversation.pinned, conversation.muted])
   952→
   953→  const startPress = () => {
   954→    pressTimer.current = setTimeout(() => setMenuOpen(true), 500)
   955→  }
   956→  const cancelPress = () => {
   957→    if (pressTimer.current) clearTimeout(pressTimer.current)
   958→  }
   959→
   960→  return (
   961→    <div className="relative">
   962→      <button
   963→        onClick={onOpen}
   964→        onPointerDown={startPress}
   965→        onPointerUp={cancelPress}
   966→        onPointerLeave={cancelPress}
   967→        className={cn(
   968→          'flex w-full items-center gap-3 rounded-xl p-2.5 text-left transition-colors hover:bg-accent/50',
   969→          pinned && 'bg-primary/5',
   970→          archived && 'opacity-70'
   971→        )}
   972→      >
   973→        <UserAvatar name={conversation.name} src={conversation.photoUrl} size={48} online={online} />
   974→        <div className="min-w-0 flex-1">
   975→          <div className="flex items-baseline justify-between gap-2">
   976→            <span className="flex min-w-0 items-center gap-1 truncate text-sm font-semibold">
   977→              <span className="truncate">{conversation.name}</span>
   978→              {pinned && <Pin size={11} className="shrink-0 text-primary" />}
   979→              {muted && <BellOff size={11} className="shrink-0 text-muted-foreground" />}
   980→              {archived && <Archive size={11} className="shrink-0 text-muted-foreground" />}
   981→            </span>
   982→            {conversation.lastMessage && (
   983→              <span className="shrink-0 text-[10px] text-muted-foreground">
   984→                {timeAgo(conversation.lastMessage.createdAt)}
   985→              </span>
   986→            )}
   987→          </div>
   988→          {isTyping ? (
   989→            // Round 7: typing-status takes priority over both draft +
   990→            // last-message preview — it's the most time-sensitive info
   991→            // ("the peer is about to send a message"). Unread badge still
   992→            // shows alongside.
   993→            <div className="mt-0.5 flex items-center justify-between gap-2">
   994→              <span className="truncate text-xs italic text-primary">typing…</span>
   995→              {conversation.unread > 0 && (
   996→                <span
   997→                  className={cn(
   998→                    'grid h-5 min-w-5 shrink-0 place-items-center rounded-full px-1.5 text-[10px] font-bold',
   999→                    muted ? 'bg-muted text-muted-foreground' : 'bg-primary text-primary-foreground'
  1000→                  )}
  1001→                >
  1002→                  {conversation.unread > 9 ? '9+' : conversation.unread}
  1003→                </span>
  1004→              )}
  1005→            </div>
  1006→          ) : draftPreview ? (
  1007→            // Draft indicator takes priority over the last-message preview so
  1008→            // the user sees their unsaved text at a glance.
  1009→            <div className="mt-0.5 flex items-center justify-between gap-2">
  1010→              <span className="flex min-w-0 items-center gap-1 truncate text-xs text-primary">
  1011→                <Pencil size={11} className="shrink-0" />
  1012→                <span className="truncate">Draft: {draftPreview}</span>
  1013→              </span>
  1014→              {conversation.unread > 0 && (
  1015→                <span
  1016→                  className={cn(
  1017→                    'grid h-5 min-w-5 shrink-0 place-items-center rounded-full px-1.5 text-[10px] font-bold',
  1018→                    muted ? 'bg-muted text-muted-foreground' : 'bg-primary text-primary-foreground'
  1019→                  )}
  1020→                >
  1021→                  {conversation.unread > 9 ? '9+' : conversation.unread}
  1022→                </span>
  1023→              )}
  1024→            </div>
  1025→          ) : (
  1026→            <div className="flex items-center justify-between gap-2">
  1027→              <span
  1028→                className={cn(
  1029→                  'truncate text-xs',
  1030→                  conversation.unread > 0 && !muted ? 'font-medium text-foreground' : 'text-muted-foreground'
  1031→                )}
  1032→              >
  1033→                {lastMessagePreview(conversation, meId)}
  1034→              </span>
  1035→              {conversation.unread > 0 && (
  1036→                <span
  1037→                  className={cn(
  1038→                    'grid h-5 min-w-5 shrink-0 place-items-center rounded-full px-1.5 text-[10px] font-bold',
  1039→                    muted ? 'bg-muted text-muted-foreground' : 'bg-primary text-primary-foreground'
  1040→                  )}
  1041→                >
  1042→                  {conversation.unread > 9 ? '9+' : conversation.unread}
  1043→                </span>
  1044→              )}
  1045→            </div>
  1046→          )}
  1047→        </div>
  1048→      </button>
  1049→      {menuOpen && (
  1050→        <>
  1051→          <div className="fixed inset-0 z-20" onClick={() => setMenuOpen(false)} />
  1052→          <div className="absolute right-2 top-12 z-30 w-44 overflow-hidden rounded-xl border border-border bg-popover p-1 shadow-xl fc-rise">
  1053→            <button
  1054→              onClick={() => { onTogglePin(); setPinned((p) => !p); setMenuOpen(false) }}
  1055→              className="flex w-full items-center gap-2 rounded-lg px-3 py-2 text-sm hover:bg-accent"
  1056→            >
  1057→              <Pin size={14} /> {pinned ? 'Unpin chat' : 'Pin chat'}

  1516→          m.id === messageId

  1518→                ...m,
  1519→                content,
  1520→                edited: true,
  1521→                editedAt: editedAt || m.editedAt || new Date().toISOString(),
  1522→              }
  1523→            : m
  1524→        )
  1525→      )

  1528→    // Group typing (sender-aware): server emits group:typing:indicator
  1529→    // { groupId, userId, isTyping } to all other group members (excludes the
  1530→    // typer). We filter by groupId + ignore our own id (defensive) and track
  1531→    // the set of userIds currently typing so the indicator can show "Aria is
  1532→    // typing…" or "3 people are typing…".
  1533→    const onGroupTyping = ({
  1534→      groupId,
  1535→      userId,
  1536→      isTyping,
  1537→    }: {

  1539→      userId: string
  1540→      isTyping: boolean

  1550→    }
  1551→
  1552→    socket.on(RT.MESSAGE_NEW, onNew)
  1553→    socket.on(RT.GROUP_MESSAGE_NEW, onNew)
